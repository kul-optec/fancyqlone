cmake_minimum_required(VERSION 4.1)

set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

project(cyqlone-benchmarks LANGUAGES CXX)

option(WITH_HPIPM "Enable the HPIPM benchmarks and tests" ON)
option(WITH_PIQP "Enable the PIQP benchmarks and tests" ON)
option(WITH_TREEQP "Enable the TreeQP benchmarks and tests" ON)

option(BUILD_TESTING "Build tests" OFF)

include(CTest)

# -----------------------------------------------------------------------------
# Dependencies
# -----------------------------------------------------------------------------

find_package(CLI11 CONFIG REQUIRED)
find_package(benchmark CONFIG REQUIRED)
find_package(fmt CONFIG REQUIRED)
find_package(cyqlone CONFIG COMPONENTS QPALM Extra REQUIRED)
if(WITH_HPIPM)
    find_package(hpipm CONFIG REQUIRED)
endif()
if(WITH_PIQP)
    find_package(blasfeo CONFIG REQUIRED)

    if(TARGET blasfeo AND NOT TARGET blasfeo::blasfeo)
        add_library(blasfeo::blasfeo INTERFACE IMPORTED)

        target_link_libraries(
            blasfeo::blasfeo
            INTERFACE
                blasfeo
        )
    endif()

    find_package(piqp CONFIG REQUIRED)
endif()
if(WITH_TREEQP)
    find_package(treeqp CONFIG REQUIRED)
endif()

# -----------------------------------------------------------------------------
# Benchmarks
# -----------------------------------------------------------------------------

add_executable(spring-mass
    spring-mass.cpp
    benchmark-reporter.cpp
)

target_link_libraries(spring-mass PRIVATE
    benchmark::benchmark
    CLI11::CLI11
    fmt::fmt
    cyqlone::qpalm
    cyqlone::example-problems
)

add_executable(spring-mass2
    spring-mass2.cpp
    benchmark-reporter.cpp
)

target_link_libraries(spring-mass2 PRIVATE
    benchmark::benchmark
    CLI11::CLI11
    fmt::fmt
    cyqlone::qpalm
    cyqlone::example-problems
)

add_executable(centroidal-quadruped
    centroidal-quadruped.cpp
    benchmark-reporter.cpp
)

target_link_libraries(centroidal-quadruped PRIVATE
    benchmark::benchmark
    CLI11::CLI11
    fmt::fmt
    cyqlone::qpalm
    cyqlone::example-problems
)

if(WITH_HPIPM)
    target_sources(spring-mass PRIVATE
        treehpipm.cpp
    )

    target_link_libraries(spring-mass PRIVATE
        hpipm
    )

    target_compile_definitions(spring-mass PRIVATE
        WITH_HPIPM=1
    )

    target_sources(spring-mass2 PRIVATE
        treehpipm.cpp
    )

    target_link_libraries(spring-mass2 PRIVATE
        hpipm
    )

    target_compile_definitions(spring-mass2 PRIVATE
        WITH_HPIPM=1
    )

    target_sources(centroidal-quadruped PRIVATE
        treehpipm.cpp
    )

    target_link_libraries(centroidal-quadruped PRIVATE
        hpipm
    )

    target_compile_definitions(centroidal-quadruped PRIVATE
        WITH_HPIPM=1
    )
endif()

if(WITH_PIQP)
    target_sources(spring-mass PRIVATE
        mspiqp.cpp
    )

    target_link_libraries(spring-mass PRIVATE
        piqp::piqp
    )

    target_compile_definitions(spring-mass PRIVATE
        WITH_PIQP=1
    )

    target_sources(spring-mass2 PRIVATE
        mspiqp.cpp
    )

    target_link_libraries(spring-mass2 PRIVATE
        piqp::piqp
    )

    target_compile_definitions(spring-mass2 PRIVATE
        WITH_PIQP=1
    )

    target_sources(centroidal-quadruped PRIVATE
        mspiqp.cpp
    )

    target_link_libraries(centroidal-quadruped PRIVATE
        piqp::piqp
    )

    target_compile_definitions(centroidal-quadruped PRIVATE
        WITH_PIQP=1
    )
endif()
if(WITH_TREEQP)
    target_sources(spring-mass PRIVATE
        treeqp.cpp
    )

    target_link_libraries(spring-mass PRIVATE
        treeqp::treeqp 
    )

    target_compile_definitions(spring-mass PRIVATE
        WITH_TREEQP=1
    )

    target_sources(spring-mass2 PRIVATE
        treeqp.cpp
    )

    target_link_libraries(spring-mass2 PRIVATE
        treeqp::treeqp
    )

    target_compile_definitions(spring-mass2 PRIVATE
        WITH_TREEQP=1
    )
endif()
# -----------------------------------------------------------------------------
# Tests
# -----------------------------------------------------------------------------

if(BUILD_TESTING AND WITH_HPIPM)
    # CONFIG is important: use the package configuration generated by Conan.
    find_package(GTest CONFIG REQUIRED)

    include(GoogleTest)

    add_executable(testing
        test.cpp
        treehpipm.cpp
    )

    target_compile_features(testing PRIVATE
        cxx_std_20
    )

    target_compile_definitions(testing PRIVATE
        WITH_HPIPM=1
    )

    target_link_libraries(testing PRIVATE
        GTest::gtest
        GTest::gtest_main
        cyqlone::qpalm
        cyqlone::example-problems
        hpipm
    )

    gtest_discover_tests(
        testing
        DISCOVERY_TIMEOUT 30
    )
endif()
