18/// Storage for a linear-quadratic OCP with the initial states x₀ eliminated.
19/// ~~~
20/// ₙ₋₁
21/// minimize ∑ [½ uᵢᵀ Rᵢ uᵢ + uᵢᵀ Sᵢ xᵢ + ½ xᵢᵀ Qᵢ xᵢ + rᵢᵀuᵢ + qᵢᵀxᵢ]
22/// ⁱ⁼¹
23/// + ½ u₀ᵀ R₀ u₀ + (r₀ + S₀ xᵢₙᵢₜ)ᵀ u₀
24/// + ½ xₙᵀ Qₙ xₙ + qₙᵀ xₙ
25/// s.t. xᵢ₊₁ = Aᵢ xᵢ + Bᵢ uᵢ + cᵢ
26/// lᵢ ≤ Cᵢ xᵢ + Dᵢ uᵢ ≤ uᵢ
27/// l₀ - C₀ xᵢₙᵢₜ ≤ D₀ U₀ ≤ u₀ - C₀ xᵢₙᵢₜ
28/// lₙ ≤ Cₙ xₙ ≤ uₙ
29/// ~~~
30/// The matrices are combined per stage, with inputs ordered first.
31/// The first and last stage are special because of the lack of x₀ and uₙ.
32/// ~~~
33/// Hᵢ = [ Rᵢ Sᵢ ], H₀ = [ R₀ 0 ], Fᵢ = [ Bᵢ Aᵢ ], Gᵢ = [ Dᵢ Cᵢ ], G₀ = [ D₀ Cₙ ]
34/// [ Sᵢᵀ Qᵢ ] [ 0 Qₙ ]
35/// ~~~
36/// Due to the elimination of x₀, there may be fewer constraints for the first stage, which is
37/// tracked by the Ju0 mask. When reconstructing the solution, the multipliers for eliminated
38/// constraints are set to zero (we assume that the initial state is feasible w.r.t. the state
39/// constraints).
40/// @ingroup topic-ocp-formulations
41template <class T = real_t>
42struct CyqloneStorage {
43 using value_type = T;
44 index_t N_horiz;
45 index_t nx, nu, ny, ny_0, ny_N;
46 std::vector<bool> Ju0;
47 using matrix = batmat::matrix::Matrix<value_type, index_t>;
48 matrix data_H = [this] {
49 return matrix{{.depth = N_horiz, .rows = nu + nx, .cols = nu + nx}};
50 }();
51 matrix data_F = [this] { return matrix{{.depth = N_horiz, .rows = nx, .cols = nu + nx}}; }();
52 matrix data_G = [this] {
53 return matrix{{.depth = N_horiz - 1, .rows = ny, .cols = nu + nx}};
54 }();
55 matrix data_G0N = [this] {
56 return matrix{{.depth = 1, .rows = ny_0 + ny_N, .cols = nu + nx}};
57 }();
58 matrix data_rq = [this] { return matrix{{.depth = N_horiz, .rows = nu + nx, .cols = 1}}; }();
59 matrix data_c = [this] { return matrix{{.depth = N_horiz, .rows = nx, .cols = 1}}; }();
60 matrix data_lb = [this] { return matrix{{.depth = N_horiz - 1, .rows = ny, .cols = 1}}; }();
61 matrix data_lb0N = [this] { return matrix{{.depth = 1, .rows = ny_0 + ny_N, .cols = 1}}; }();
62 matrix data_ub = [this] { return matrix{{.depth = N_horiz - 1, .rows = ny, .cols = 1}}; }();
63 matrix data_ub0N = [this] { return matrix{{.depth = 1, .rows = ny_0 + ny_N, .cols = 1}}; }();
64 std::vector<index_t> indices_G0 = std::vector<index_t>(ny_0);
