163template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
164void SerialFanCyqloneSolver<VL, T, DefaultOrder, Ctx>::pack_variables(
165 std::span<const value_type> ux_lin, FanStorage &ux) const {
166 BATMAT_ASSERT(static_cast<index_t>(ux_lin.size()) == num_variables());
167 BATMAT_ASSERT(static_cast<index_t>(ux.branches.size()) == num_branches);
168 index_t offset = 0;
169 for (index_t c = 0; c < num_branches; ++c) {
170 const auto &solver = solver_storages[c].cyqlone_solver;
171 const index_t nux = nu + solver.nx, N_horiz = solver.N_horiz;
172
173 BATMAT_ASSERT(ux.branches[c].depth() == solver.ceil_N());
174 BATMAT_ASSERT(ux.branches[c].rows() == nux);
175 BATMAT_ASSERT(ux.branches[c].cols() == 1);
176 // Match the compact export stride, excluding each branch's dummy u0.
177 solver.pack_variables(ux_lin.subspan(offset, nux * N_horiz), ux.branches[c]);
178 const auto root = solver.inverse_stage_index(0);
179 ux.branches[c].batch(root.c * solver.n + root.i)(root.l).top_rows(nu).set_constant(0);
180 offset += nux * N_horiz - nu;
181 }
182 using crview = guanaqo::MatrixView<const value_type, index_t>;
183 la::copy(crview::as_column(ux_lin.first(nu)), ux.first.batch(0));
184}
185template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
186void SerialFanCyqloneSolver<VL, T, DefaultOrder, Ctx>::unpack_variables(
187 const FanStorage &ux, std::span<value_type> ux_lin) const {
188 BATMAT_ASSERT(static_cast<index_t>(ux.branches.size()) == num_branches);
189
190 index_t offset = static_cast<index_t>(ux_lin.size()); //put u0_sol at the end of ux_lin
191
192 // Copy u0_sol into ux_lin
193 using rview = guanaqo::MatrixView<value_type, index_t>;
194
195 // copy the rest of the variables from each branch into ux_lin but leave out the dummy u0
196 for (index_t c = num_branches - 1; c >= 0; --c) {
197 index_t nux = nu + solver_storages[c].cyqlone_solver.nx;
198 index_t ceil_N = solver_storages[c].cyqlone_solver.ceil_N();
199 index_t N_horiz = solver_storages[c].cyqlone_solver.N_horiz;
200
201 BATMAT_ASSERT(ux.branches[c].depth() == ceil_N);
202 BATMAT_ASSERT(ux.branches[c].rows() == nux);
203 BATMAT_ASSERT(ux.branches[c].cols() == 1);
204 solver_storages[c].cyqlone_solver.unpack_variables(
205 ux.branches[c], ux_lin.subspan(offset - nux * N_horiz, nux * N_horiz));
206 offset -= nux * N_horiz - nu;
207 }
208 la::copy(
209 ux.first.batch(0).top_rows(nu),
210 rview::as_column(ux_lin.first(
211 nu))); // copy u0_sol from the first element of ux into the first nu elements of ux_lin
212}
