109qp::problems::SpringMassProblem create_problem(const SpringMassParams &params) {
110 auto problem = qp::problems::spring_mass(params);
111 // Add a random disturbance to the initial position of the masses
112 std::seed_seq seed{params.seed, params.seed >> 32};
113 std::mt19937 rng{seed};
114 using dist_t = std::uniform_real_distribution<real_t>;
115 auto dist = params.p_max > 1 ? dist_t{-3, 3} : dist_t{-0.6, -0.1};
116 for (index_t r = 0; r < static_cast<index_t>(params.masses.size()); ++r)
117 problem.ocp.b(0)(r, 0) += dist(rng);
118 return problem;
119}

176template <index_t VL, qp::StorageOrder Order>
177void run_benchmark(benchmark::State &state, [[maybe_unused]] const std::string &param_name,
178 const SpringMassParams &params, qp::CyQPALMBackendSettings backend_settings,
179 qp::Settings settings, bool warm = false, bool trace = false) try {
180 disable_tracing(); // No tracing during the actual benchmark
181 if (backend_settings.processors < 1)
182 return state.SkipWithMessage("Number of processors must be at least 1.");
183 auto problem = create_problem(params);
184 // Build a QPALM Cyqlone solver
185 auto ocp = fancyqlone::CyqloneStorage<>::build(problem.ocp);
186 auto backend = qp::make_cyqpalm_backend<VL, Order>(ocp, {}, backend_settings);
187 qp::Solver<qp::CyQPALMBackend<VL, Order> *> qpalm{backend.get(), settings};
188 // Main benchmark
189 qp::SolverStatus status;
190 double time_solve = 0, time_active_set_change = 0, time_line_search = 0,
191 time_recompute_outer = 0, time_recompute_inner = 0, time_mat_vec = 0;
192 qpalm(); // Warm up
193 if (warm) {
194 const auto nx = problem.ocp.dim.nx, nu = problem.ocp.dim.nu;
195 qpalm.warm_start_solution();
196 auto x = qpalm.get_solution();
197 std::copy_n(x.data() + nu, nx, problem.ocp.b(0).data);
198 ocp.update(problem.ocp);
199 update_cyqpalm_backend(*qpalm.backend, ocp);
200 }
201 for (auto _ : state) {
202 status = qpalm();
203 state.SetIterationTime(seconds(qpalm.stats->timings.total.wall_time).count());
204 time_solve += seconds(qpalm.stats->timings.solve.wall_time).count();
205 time_active_set_change += seconds(qpalm.stats->timings.active_set_change.wall_time).count();
206 time_line_search += seconds(qpalm.stats->timings.line_search.wall_time).count();
207 time_recompute_outer += seconds(qpalm.stats->timings.recompute_outer.wall_time).count();
208 time_recompute_inner += seconds(qpalm.stats->timings.recompute_inner.wall_time).count();
209 time_mat_vec += seconds(qpalm.stats->timings.mat_vec_A.wall_time).count();
210 time_mat_vec += seconds(qpalm.stats->timings.mat_vec_AT.wall_time).count();
211 time_mat_vec += seconds(qpalm.stats->timings.mat_vec_M.wall_time).count();
212 time_mat_vec += seconds(qpalm.stats->timings.mat_vec_MT.wall_time).count();
213 time_mat_vec += seconds(qpalm.stats->timings.mat_vec_Q.wall_time).count();
214 }
215 if (trace)
216 trace_run([&] { qpalm(); }, param_name);
217 auto sol = ocp.reconstruct_solution(problem.ocp, qpalm.get_solution(),
218 qpalm.get_inequality_multipliers(),
219 qpalm.get_equality_multipliers());
220 auto kkt_error = problem.ocp.compute_kkt_error(sol);
221
222 state.counters["status"] = counter(status);
223 state.counters["success"] = counter(status == qp::SolverStatus::Converged);
224 state.counters["iter"] = counter(qpalm.stats->inner_iter);
225 state.counters["outer_iter"] = counter(qpalm.stats->outer_iter);
226 state.counters["num_factor"] = counter(qpalm.stats_backend->num_factor);
227 state.counters["num_upd"] = counter(qpalm.stats_backend->num_updates);
228 state.counters["rank_upd"] = counter(qpalm.stats_backend->rank_updates);
229 state.counters["t_solve"] = counter_avg(time_solve);
230 state.counters["t_active_set_change"] = counter_avg(time_active_set_change);
231 state.counters["t_line_search"] = counter_avg(time_line_search);
232 state.counters["t_recompute_outer"] = counter_avg(time_recompute_outer);
233 state.counters["t_recompute_inner"] = counter_avg(time_recompute_inner);
234 state.counters["t_mat_vec"] = counter_avg(time_mat_vec);
235 state.counters["res_dual"] = counter(kkt_error.stationarity);
236 state.counters["res_eq"] = counter(kkt_error.equality_residual);
237 state.counters["res_ineq"] = counter(kkt_error.inequality_residual);
238 state.counters["compl"] = counter(kkt_error.complementarity);
239} catch (const std::exception &e) {
240 state.SkipWithError(e.what());
241}

251void run_benchmark_hpipm(benchmark::State &state, const SpringMassParams &params,
252 WarmStartHPIPM warm = WarmStartHPIPM::NoWarmStart) {
253 auto problem = create_problem(params);
254 auto qp_data = build_hpipm_qp(problem.ocp);
255 auto solver_data = create_hpipm_solver(qp_data, warm != WarmStartHPIPM::NoWarmStart ? 2 : 0);
256 // Main benchmark
257 solve_hpipm(*qp_data, *solver_data); // Warm up
258 std::optional<fancyqlone::LinearOCPStorage::Solution> prev_sol;
259 if (warm != WarmStartHPIPM::NoWarmStart) {
260 prev_sol = get_solution_hpipm(*solver_data);
261 update_x0_hpipm(problem.ocp, *qp_data, prev_sol->solution);
262 if (warm == WarmStartHPIPM::WarmZero)
263 std::ranges::fill(prev_sol->solution, real_t{0});
264 else if (warm == WarmStartHPIPM::WarmShift)
265 shift_solution_hpipm(*qp_data, prev_sol->solution);
266 }
267 for (auto _ : state) {
268 if (prev_sol)
269 warm_start_hpipm(*solver_data, prev_sol->solution);
270 auto time_solve = solve_hpipm(*qp_data, *solver_data);
271 state.SetIterationTime(seconds(time_solve).count());
272 }
273 auto sol = get_solution_hpipm(*solver_data);
274 auto kkt_error = problem.ocp.compute_kkt_error(sol);
275 auto stats = get_stats_hpipm(*solver_data);
276 state.counters["status"] = counter(stats.status);
277 state.counters["success"] = counter(stats.status == 0);
278 state.counters["iter"] = counter(stats.iter);
279 state.counters["max_res_stat"] = counter(stats.max_res_stat);
280 state.counters["max_res_eq"] = counter(stats.max_res_eq);
281 state.counters["max_res_ineq"] = counter(stats.max_res_ineq);
282 state.counters["max_res_comp"] = counter(stats.max_res_comp);
283 state.counters["res_dual"] = counter(kkt_error.stationarity);
284 state.counters["res_eq"] = counter(kkt_error.equality_residual);
285 state.counters["res_ineq"] = counter(kkt_error.inequality_residual);
286 state.counters["compl"] = counter(kkt_error.complementarity);
287}

295std::generator<Problem> get_spring_mass_params(const Options &opts) {
296 using std::format;
297 for (auto M : opts.masses)
298 for (auto N : opts.horizon)
299 for (uint64_t seed = opts.seed; seed < opts.seed + opts.num_instances; ++seed)
300 switch (opts.problem_type) {
301 case ProblemType::WangBoyd2008:
302 co_yield {
303 .name = format("spring-mass(M={},N={},seed={}) wb2008", M, N, seed),
304 .params = SpringMassParams::wang_boyd_2008(M, N, seed),
305 };
306 break;
307 case ProblemType::WangBoyd2008Width:
308 co_yield {
309 .name = format("spring-mass(M={},N={},seed={}) wb2008w", M, N, seed),
310 .params = SpringMassParams::wang_boyd_2008_width(M, N, seed, 1),
311 };
312 break;
313 case ProblemType::Domahidi2012:
314 co_yield {
315 .name = format("spring-mass(M={},N={},seed={}) d2012", M, N, seed),
316 .params = SpringMassParams::domahidi_2012(M, N, seed),
317 };
318 break;
319 case ProblemType::ActiveStateConstr:
320 co_yield {
321 .name = format("spring-mass(M={},N={},seed={}) act", M, N, seed),
322 .params = SpringMassParams::active_state_constr(M, N, seed),
323 };
324 break;
325 default: throw std::runtime_error("Unknown problem type");
326 }
327}

351template <index_t VL, qp::StorageOrder O>
352std::generator<Solver> get_cyqlone_solvers(const Options &opts) {
353 const auto cyqlone_solver = [trace{opts.trace}](std::string_view name,
354 const qp::CyQPALMBackendSettings &backend,
355 const qp::Settings &settings) {
356 const auto param_name =
357 std::format("cyqlone(p={},v={},{},{})", backend.processors, VL, order(O), name);
358 return Solver{param_name, [=](benchmark::State &state, const SpringMassParams &params) {
359 run_benchmark<VL, O>(state, param_name, params, backend, settings, true,
360 trace);
361 }};
362 };
363 for (auto p : opts.parallelism) {
364 qp::CyQPALMBackendSettings backend{
365 .processors = p,
366 .changing_constr_factor = opts.changing_constr_factor,
367 .max_update_count = 20,
368 .tricyqle_params =
369 {
370 .pcr_max_update_fraction = opts.pcr_max_update_fraction,
371 .cr_max_update_fraction_Y0 = opts.cr_max_update_fraction,
372 .parallel_solve_cr_threshold = opts.parallel_solve_cr_threshold,
373 .parallel_factor_pcr_threshold = opts.parallel_factor_pcr_threshold,
374 },
375 };
376 qp::Settings settings{
377 .tolerance = 1e-8,
378 .dual_tolerance = 1e-8,
379 .initial_penalty_y = 20,
380 .verbose = false,
381 };
382 qp::CyQPALMBackendSettings backend_no_upd = backend;
383 backend_no_upd.max_update_count = 0;
384 qp::Settings settings_warm = settings;
385 settings_warm.initial_penalty_y = 1e4;
386 settings_warm.initial_inner_tolerance = 1e-4;
387 if (opts.pcr)
388 backend.tricyqle_params.solve_method = fancyqlone::SolveMethod::PCR;
389 if (opts.cold) {
390 backend.strategy = backend_no_upd.strategy = qp::WarmStartingStrategy::Zeros;
391 co_yield cyqlone_solver("zero", backend, settings);
392 if (opts.no_updates)
393 co_yield cyqlone_solver("zero,upd=0", backend_no_upd, settings);
394 }
395 if (opts.warm_shift) {
396 backend.strategy = backend_no_upd.strategy = qp::WarmStartingStrategy::Shift;
397 co_yield cyqlone_solver("shift", backend, settings_warm);
398 if (opts.no_updates)
399 co_yield cyqlone_solver("shift,upd=0", backend_no_upd, settings_warm);
400 }
401 if (opts.warm_copy) {
402 backend.strategy = backend_no_upd.strategy = qp::WarmStartingStrategy::Copy;
403 co_yield cyqlone_solver("copy", backend, settings_warm);
404 if (opts.no_updates)
405 co_yield cyqlone_solver("copy,upd=0", backend_no_upd, settings_warm);
406 }
407 }
408}

433template <qp::StorageOrder Order>
434std::generator<Solver> get_cyqlone_solvers_vl(const Options &opts) {
435 static constexpr auto VLs = batmat::types::vl_for_real_t;
436 static constexpr auto to_string = [](auto vl) { return std::to_string(vl); };
437 static constexpr auto lut = guanaqo::make_lut<VLs>([]<index_t VL>(batmat::index_constant<VL>) {
438 return std::make_pair(VL, get_cyqlone_solvers<VL, Order>);
439 });
440 for (auto v : opts.vector_length) {
441 auto dispatch = std::ranges::find(lut, v, &decltype(lut)::value_type::first);
442 if (dispatch == lut.end())
443 throw std::invalid_argument("Unsupported vector length. Supported lengths: " +
444 guanaqo::join(std::views::transform(VLs, to_string)));
445 co_yield elements_of(dispatch->second(opts));
446 }
447}

457auto register_benchmarks(const Options &opts) {
458 size_t max_problem_name_len = 0;
459 size_t max_solver_name_len = 0;
460 for (auto problem : get_spring_mass_params(opts)) {
461 max_problem_name_len = std::max(max_problem_name_len, problem.name.size());
462 for (auto solver : get_solvers(opts)) {
463 max_solver_name_len = std::max(max_solver_name_len, solver.name.size());
464 benchmark::RegisterBenchmark(std::format("{}@{}", problem.name, solver.name),
465 [params = problem.params, run = std::move(solver.run)](
466 benchmark::State &state) { run(state, params); })
467 ->MeasureProcessCPUTime()
468 ->UseManualTime()
469 ->Unit(benchmark::kMillisecond)
470 ->ComputeStatistics("max", [](auto &v) { return *std::ranges::max_element(v); })
471 ->ComputeStatistics("min", [](auto &v) { return *std::ranges::min_element(v); })
472 ->GetName();
473 }
474 }
475 return std::make_pair(max_problem_name_len, max_solver_name_len);
476}
