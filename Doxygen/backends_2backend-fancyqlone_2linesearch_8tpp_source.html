19template <class T, size_t N>
20void CyQPALMBackend<VL, DefaultOrder, LinearSolver>::merge_chunk(
21 std::span<const T> chunk, size_t chunk_index, std::span<const std::array<size_t, N>> separators,
22 std::span<T> out) {
23 GUANAQO_TRACE("merge_chunk", 0, chunk.size());
24 size_t num_chunks = separators.size();
25 BATMAT_ASSUME(chunk_index < num_chunks);
26 std::array<size_t, N> offsets{};
27 for (size_t i = 0; i < N; ++i)
28 for (size_t c = 0; c < chunk_index; ++c)
29 offsets[i] += separators[c][i];
30 for (size_t i = 0; i < N - 1; ++i)
31 for (size_t c = chunk_index; c < num_chunks; ++c)
32 offsets[i + 1] += separators[c][i];
33 std::copy(chunk.begin(), chunk.begin() + separators[chunk_index][0], out.begin() + offsets[0]);
34 for (size_t i = 1; i < N; ++i)
35 std::copy(chunk.begin() + separators[chunk_index][i - 1],
36 chunk.begin() + separators[chunk_index][i], out.begin() + offsets[i]);
37}

39template <index_t VL, StorageOrder DefaultOrder, class LinearSolver>
40BreakpointsResult CyQPALMBackend<VL, DefaultOrder, LinearSolver>::compute_partition_breakpoints_cyq(
41 Context &ctx, std::vector<Breakpoint> &breakpoints, const ineq_constr_vec_t &Σ,
42 const ineq_constr_vec_t &y, const ineq_constr_vec_t &Ad, const ineq_constr_vec_t &Ax,
43 const ineq_constr_vec_t &b_min, const ineq_constr_vec_t &b_max) {
44 auto t = get_timed(&Timings::breakpoints);
45 using std::isfinite;
46 using std::sqrt;
47 // Allocate memory
48 const index_t ny_M = std::max(ocp.ny, ocp.ny_0 + ocp.ny_N);
49 ctx.run_single_sync([&] {
50 const index_t m = ocp.ceil_N() * ny_M;
51 breakpoints.resize(2 * m);
52 breakpoints_temp.resize(2 * m);
53 thread_indices.resize(ocp.p);
54 });
55 // Parallelization and vectorization
56 auto thr_parts = std::span{thread_indices}.subspan(0, ocp.p);
57 const index_t ti = ocp.riccati_thread_assignment(ctx);
58 const index_t bpt_per_thr = 2 * ny_M * ocp.n * ocp.v;
59 // Partition the breakpoints into a finite and an infinite part (per thread)
60 Breakpoint *const fin_0 = breakpoints_temp.data() + ti * bpt_per_thr;
61 Breakpoint *const inf_0 = fin_0 + bpt_per_thr;
62 Breakpoint *fin = fin_0, *inf = inf_0;
63 // Compute break points t[i] and intermediate values α[i] and δ[i]
64 const auto brkpts_simd = [&](auto Σi, auto yi, auto Adi, auto Axi, auto li, auto ui) {
65 const auto s = sqrt(Σi);
66 const auto δ2 = s * Adi, δ1 = -δ2;
67 const auto α1 = (yi + Σi * (Axi - li)) / s, α2 = (Σi * (ui - Axi) - yi) / s;
68 const auto t1 = α1 / δ1, t2 = α2 / δ2;
69 BATMAT_FULLY_UNROLLED_FOR (int l = 0; l < ocp.v; ++l) {
70 *(isfinite(t1[l]) ? fin++ : --inf) = {.t = t1[l], .δ = δ1[l]};
71 *(isfinite(t2[l]) ? fin++ : --inf) = {.t = t2[l], .δ = δ2[l]};
72 }
73 // Invariant: finite values in [fin_0, fin) and infinite values in [inf, inf_0)
74 };
75 const auto brkpts_batch = [&](auto Σj, auto yj, auto Adj, auto Axj, auto b_min_j,
76 auto b_max_j) {
77 linalg::for_each_elementwise(brkpts_simd, Σj, yj, Adj, Axj, b_min_j, b_max_j);
78 };
79 ocp.foreach_batch(ctx, brkpts_batch, Σ, y, Ad, Ax, b_min, b_max);
80 // Now partition the finite breakpoints into negative and positive parts.
81 // Partitioning the chunk of each thread separately improves partitioning performance
82 // later on in the line search because of branch prediction.
83 auto [pos, large] = [&] {
84 GUANAQO_TRACE("linesearch breakpoints cyqlone partition", ti);
85 auto pos = partition(fin_0, fin, [](Breakpoint p) { return p.t <= 0; }).begin();
86 auto large = partition(pos, fin, [](Breakpoint p) { return p.t <= 1; }).begin();
87 return std::pair{pos, large};
88 }();
89 // Store the separator indices for all threads, to merge the partitions in parallel later.
90 thr_parts[ti][0] = pos - fin_0; // index of first positive breakpoint (per thread)
91 thr_parts[ti][1] = large - fin_0; // index of first breakpoint larger than 1 (per thread)
92 thr_parts[ti][2] = fin - fin_0; // index of first infinite breakpoint (per thread)
93 thr_parts[ti][3] = inf_0 - fin_0; // total number of breakpoints (per thread)
94 auto thr_parts_done = ctx.arrive(); // all-to-all
95 // Compute the partial sums
96 PartitionedBreakpoints pos_neg_bp{.neg_bp = std::span{fin_0, pos},
97 .pos_bp = std::span{pos, fin}};
98 auto ab_neg = partial_sum_negative(pos_neg_bp);
99 // Synchronize the separator indices for all threads.
100 ctx.wait(std::move(thr_parts_done));
101 // Merge all local partitions of all threads into a single partitioned array.
102 GUANAQO_TRACE("linesearch breakpoints cyqlone merge", ti);
103 merge_chunk<Breakpoint, 4>(std::span{fin_0, inf_0}, ti, thr_parts, std::span{breakpoints});
104 // Compute the total sums across all threads.
105 auto ab_neg_and_merge_done = ctx.arrive_reduce(ab_neg, std::plus<>{});
106 // Compute the final partition indices by summing the partition sizes of all threads.
107 auto first_pos = std::accumulate(thr_parts.begin(), thr_parts.end(), breakpoints.begin(),
108 [](auto it, auto &i) { return it += i[0]; }),
109 first_inf = std::accumulate(thr_parts.begin(), thr_parts.end(), breakpoints.begin(),
110 [](auto it, auto &i) { return it += i[2]; });
111 // Wait for the total sums across all threads. Also synchronize the merged breakpoints.
112 ab_neg = ctx.wait_reduce(std::move(ab_neg_and_merge_done));
113 return {.bp = {.neg_bp = std::span{breakpoints.begin(), first_pos},
114 .pos_bp = std::span{first_pos, first_inf}},
115 .ab_neg = ab_neg};
116}
