18 using guanaqo::as_view;
19 const auto inf = std::numeric_limits<real_t>::infinity();

21 auto n_masses = static_cast<index_t>(p.masses.size());
22 const index_t ny_x = p.v_max > 0 ? 2 * n_masses : n_masses,
23 ny_x_N = p.v_max_f > 0 ? 2 * n_masses : n_masses;
24 LinearOCPStorage ocp{.dim{
25 .N_horiz = p.N_horiz,
26 .nx = 2 * n_masses, // position + velocity for each mass
27 .nu = p.n_actuators,
28 .ny = ny_x + p.n_actuators,
29 .ny_N = ny_x_N,
30 }};
31 auto [N, nx, nu, ny, ny_N] = ocp.dim;
32
33 // Continuous-time dynamics
34 eigen_mat A = eigen_mat::Zero(nx, nx), B = eigen_mat::Zero(nx, nu), b = eigen_mat::Zero(nx, 1);
35 for (Eigen::Index v = 0; v < n_masses; ++v) {
36 const auto km = p.k_spring / p.masses[v];
37 A(v, n_masses + v) = 1; // ṗ(t) = v(t)
38 A(n_masses + v, n_masses + v) = -p.friction; // v̇(t) = -ρ v(t) + ...
39 A(n_masses + v, v) = -2 * km; // ... - (2k/m) p(t) + ...
40 if (v > 0)
41 A(n_masses + v, v - 1) = km; // ... + (k/m) p_{v-1}(t)
42 if (v < n_masses - 1)
43 A(n_masses + v, v + 1) = km; // ... + (k/m) p_{v+1}(t)
44 else
45 b(n_masses + v, 0) = km * p.width; // Right wall force
46 }
47 switch (p.actuator_placement) {
48 case SpringMassParams::IndividualActuators:
49 for (Eigen::Index a = 0; a < p.n_actuators; ++a)
50 B(n_masses + a, a) = 1 / p.masses[a];
51 break;
52 case SpringMassParams::RandomActuators: {
53 std::mt19937 rng(p.seed);
54 std::vector<index_t> actuator_indices(n_masses);
55 std::ranges::iota(actuator_indices, index_t{});
56 std::shuffle(actuator_indices.begin(), actuator_indices.end(), rng);
57 for (Eigen::Index a = 0; a < p.n_actuators; ++a)
58 B(n_masses + actuator_indices[a], a) = 1 / p.masses[actuator_indices[a]];
59 } break;
60 case SpringMassParams::RandomPairsOfActuators: {
61 std::mt19937 rng(p.seed);
62 std::vector<index_t> actuator_indices_l(n_masses), actuator_indices_r(n_masses);
63 std::ranges::iota(actuator_indices_l, index_t{});
64 std::ranges::iota(actuator_indices_r, index_t{});
65 std::shuffle(actuator_indices_l.begin(), actuator_indices_l.end(), rng);
66 std::shuffle(actuator_indices_r.begin(), actuator_indices_r.end(), rng);
67 for (Eigen::Index a = 0; a < p.n_actuators; ++a) {
68 B(n_masses + actuator_indices_l[a], a) = +1 / p.masses[actuator_indices_l[a]];
69 B(n_masses + actuator_indices_r[a], a) = -1 / p.masses[actuator_indices_r[a]];
70 }
71 } break;
72 case SpringMassParams::WangBoydActuators:
73 for (Eigen::Index a = 0; a < p.n_actuators; ++a) {
74 if (a % 3 == 0) {
75 auto v_l = 2 * a, v_r = 2 * a + 1;
76 if (v_l < n_masses)
77 B(n_masses + v_l, a) = +1 / p.masses[v_l];
78 if (v_r < n_masses)
79 B(n_masses + v_r, a) = -1 / p.masses[v_r];
80 } else if (a % 3 == 1) {
81 auto v_l = 2 * a, v_r = 2 * a + 2;
82 if (v_l < n_masses)
83 B(n_masses + v_l, a) = +1 / p.masses[v_l];
84 if (v_r < n_masses)
85 B(n_masses + v_r, a) = -1 / p.masses[v_r];
86 } else {
87 auto v_l = 2 * a - 1, v_r = 2 * a + 1;
88 if (v_l < n_masses)
89 B(n_masses + v_l, a) = +1 / p.masses[v_l];
90 if (v_r < n_masses)
91 B(n_masses + v_r, a) = -1 / p.masses[v_r];
92 }
93 }
94 break;
95 default: throw std::invalid_argument("Invalid actuator placement");
96 }
97 auto Ts = p.T_horiz / static_cast<real_t>(p.N_horiz);
98 // ẋ = Ax + Bu + b → x[k+1] = Ad x[k] + Bd u[k] + bd
99 auto [Ad, Bd, bd] = discretize_zoh(A, B, b, Ts);
100
101 // Constraints rhs
102 ocp.b().set_constant(0);
103 ocp.b_min().set_constant(-inf);
104 ocp.b_max().set_constant(+inf);
105 auto x0 = ocp.b(0);
106 for (index_t v = 0; v < n_masses; ++v) // Initial positions
107 x0(v, 0) = static_cast<real_t>(v + 1) * p.width / static_cast<real_t>(n_masses + 1);
108
109 // Dynamics and constraint matrices
110 for (index_t i = 0; i < N; ++i) {
111 auto Ai = ocp.A(i), Bi = ocp.B(i), bi = ocp.b(i + 1), Ci = ocp.C(i), Di = ocp.D(i);
112 auto Qi = ocp.Q(i), Ri = ocp.R(i);
113 auto lbi = ocp.b_min(i), ubi = ocp.b_max(i);
114 for (index_t v = 0; v < n_masses; ++v) {
115 Ai = as_view(Ad);
116 Bi = as_view(Bd);
117 bi = as_view(bd);
118 Ci(v, v) = 1; // Measure displacement
119 lbi(v, 0) = x0(v, 0) + p.p_min;
120 ubi(v, 0) = x0(v, 0) + p.p_max;
121 if (p.v_max > 0) {
122 Ci(n_masses + v, n_masses + v) = 1; // Measure velocity
123 lbi(n_masses + v, 0) = -p.v_max;
124 ubi(n_masses + v, 0) = +p.v_max;
125 }
126 Qi(v, v) = p.q_pos;
127 Qi(n_masses + v, n_masses + v) = p.q_vel;
128 }
129 for (index_t a = 0; a < p.n_actuators; ++a) {
130 Di(ny_x + a, a) = 1; // Measure input
131 lbi(ny_x + a, 0) = -p.F_max;
132 ubi(ny_x + a, 0) = +p.F_max;
133 Ri(a, a) = p.r_act;
134 }
135 }
136 auto Ci = ocp.C(N), Qi = ocp.Q(N);
137 auto lbi = ocp.b_min(N), ubi = ocp.b_max(N);
138 for (index_t v = 0; v < n_masses; ++v) {
139 Ci(v, v) = 1; // Measure displacement
140 lbi(v, 0) = x0(v, 0) + p.p_min_f;
141 ubi(v, 0) = x0(v, 0) + p.p_max_f;
142 if (p.v_max_f > 0) {
143 Ci(n_masses + v, n_masses + v) = 1; // Measure velocity
144 lbi(n_masses + v, 0) = -p.v_max_f;
145 ubi(n_masses + v, 0) = +p.v_max_f;
146 }
147 Qi(v, v) = p.q_pos_f;
148 Qi(n_masses + v, n_masses + v) = p.q_vel_f;
149 }
150 std::vector<real_t> ref(nx + nu + nx);
151 for (index_t v = 0; v < n_masses; ++v) {
152 ref[v] = x0(v, 0); // position
153 ref[nx + nu + v] = x0(v, 0); // final position
154 }
155 reference_to_gradient(ocp, ref);
156
157 return {
158 .ocp = std::move(ocp),
159 .ref = std::move(ref),
160 };

163 real_t k_spring_diff) {
164 std::vector<fancyqlone::LinearOCPStorage> ocps(num_branches);
165 std::vector<real_t> ref;
166 p.k_spring -= k_spring_diff; // perturb spring constants for each branch
167 for (index_t b = 0; b < num_branches; ++b) {
168
169 // std::cout << "Branch " << b << ": k_spring = " << p.k_spring << std::endl;
170 auto ocp = ocps[b] = spring_mass(p).ocp;
171 ref = spring_mass(p).ref;
172 if (num_branches > 1) {
173 // Perturb spring constants for each branch
174 p.k_spring +=
175 2 * k_spring_diff / (num_branches - 1); // perturb spring constants for each branch
176 }
177 }
178
179 FanOCPStorage focp{.ocps = std::move(ocps)};
180 return {.ocp = std::move(focp), .ref = std::move(ref)};
