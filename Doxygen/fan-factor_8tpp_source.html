182// Algorithm 5 “Solution of a symmetric block-tridiagonal system using cyclic reduction (CR)”
183// §3.2 Cyclic reduction of block-tridiagonal linear systems
184//
185// The reverse solve routines below closely follow the structure of the corresponding factorization
186// and forward solve routines, but in reverse order. An iterative approach is used instead of
187// recursion. Note that the evaluation of λ(0) is performed during the forward solve step.
188// Depending on the problem size, either a parallel or serial version of the CR solve is used.
189
190template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
191void FanCyqloneSolver<VL, T, DefaultOrder, Ctx>::solve_reverse(Context &ctx, FanStorage &ux,
192 FanStorage &λ) {
193 GUANAQO_TRACE_REGION("fan solve reverse", 0);
194 const index_t c = ctx.total_ctx.index;
195
196 if (ctx.branch_ctx.is_master()) {
197 ctx.total_ctx.run_single_sync([&, this] { solve_reverse_u0(ux); });
198
199 solve_reverse_λ0(c, ux, λ);
200 }
201 ctx.branch_ctx.arrive_and_wait(); // wait λ()(0) to be ready
202
203 solver_storages[c].cyqlone_solver.solve_reverse(ctx.branch_ctx, ux.branches[c], λ.branches[c]);
204}
