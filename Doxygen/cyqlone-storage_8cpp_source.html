36template <class T>
37auto CyqloneStorage<T>::reconstruct_solution(const LinearOCPStorage &ocp,
38 std::span<const value_type> ux_compressed,
39 std::span<const value_type> y_compressed,
40 std::span<const value_type> λ_compressed) const
41 -> Solution {
42 Solution sol;
43 sol.solution.resize(static_cast<size_t>(N_horiz * (nu + nx) + nx));
44 sol.inequality_multipliers.resize(static_cast<size_t>(N_horiz * ny + ny_N));
45 sol.equality_multipliers.resize(static_cast<size_t>(N_horiz * nx + nx));
46 reconstruct_ineq_multipliers(y_compressed, sol.inequality_multipliers);
47 std::ranges::copy_n(ocp.b(0).data, nx, sol.solution.begin()); // x0
48 std::ranges::copy(ux_compressed, sol.solution.begin() + nx);
49 std::ranges::copy(λ_compressed, sol.equality_multipliers.begin() + nx);
50 using MatrixView = guanaqo::MatrixView<value_type, index_t>;
51 MatrixView λ0{{.data = sol.equality_multipliers.data(), .rows = nx, .cols = 1}},
52 λ1{{.data = sol.equality_multipliers.data() + nx, .rows = nx, .cols = 1}},
53 x0{{.data = sol.solution.data(), .rows = nx, .cols = 1}},
54 u0{{.data = sol.solution.data() + nx, .rows = nu, .cols = 1}};
55 λ0 = ocp.q(0);
56 guanaqo::blas::xsymv_L(value_type{1}, ocp.Q(0), x0, value_type{1}, λ0);
57 guanaqo::blas::xgemv_T(value_type{1}, ocp.S(0), u0, value_type{1}, λ0);
58 guanaqo::blas::xgemv_T(value_type{1}, ocp.A(0), λ1, value_type{1}, λ0);
59 if (ny > 0) {
60 auto y0 = MatrixView{{.data = sol.inequality_multipliers.data(), .rows = ny, .cols = 1}};
61 guanaqo::blas::xgemv_T(value_type{1}, ocp.C(0), y0, value_type{1}, λ0);
62 }
63 return sol;
64}

77index_t CyqloneStorage<T>::count_constr_0(const LinearOCPStorage &ocp, std::vector<bool> &Ju0) {
78 const auto [N, nx, nu, ny, ny_N] = ocp.dim;
79 const auto nJu0 = static_cast<index_t>(Ju0.size());
80 BATMAT_ASSUME(nJu0 == ny);

94template <class T>
95void CyqloneStorage<T>::update_impl(const LinearOCPStorage &ocp) {
96 const auto N = N_horiz;
97 // H₀ = [ R₀ 0 ]
98 // [ 0 Qₙ]
99 data_H(0).top_left(nu, nu) = ocp.R(0);
100 data_H(0).bottom_right(nx, nx) = ocp.Q(N);
101 data_H(0).bottom_left(nx, nu).set_constant(0);
102 data_H(0).top_right(nu, nx).set_constant(0);
103 // F₀ = [ B₀ 0 ]
104 data_F(0).left_cols(nu) = ocp.B(0);
105 data_F(0).right_cols(nx).set_constant(0);
106 // G₀ = [ D₀ 0 ] ny_0
107 // [ 0 Cₙ] ny_N
108 data_G0N(0).bottom_left(ny_N, nu).set_constant(0);
109 data_G0N(0).top_right(ny_0, nx).set_constant(0);
110 index_t j = 0;
111 for (index_t r = 0; r < ny; ++r) {
112 if (Ju0[r]) {
113 BATMAT_ASSUME(j < ny_0);
114 data_G0N(0).block(j, 0, 1, nu) = ocp.D(0).middle_rows(r, 1);
115 value_type t = 0;
116 for (index_t c = 0; c < nx; ++c) // lb - C₀ x₀
117 t += ocp.C(0)(r, c) * ocp.b(0)(c, 0);
118 data_lb0N(0, j, 0) = ocp.b_min(0)(r, 0) - t;
119 data_ub0N(0, j, 0) = ocp.b_max(0)(r, 0) - t;
120 indices_G0[j] = r;
121 ++j;
122 }
123 }
124 data_G0N(0).block(j, 0, ny_0 - j, nu).set_constant(0);
125 data_G0N(0).bottom_right(ny_N, nx) = ocp.C(N);
126 data_lb0N(0).bottom_rows(ny_N) = ocp.b_min().bottom_rows(ny_N);
127 data_ub0N(0).bottom_rows(ny_N) = ocp.b_max().bottom_rows(ny_N);
128 // c̃₀ = c₀ + A₀ x₀ (b_eq = [x₀, c₀, ... cₙ₋₁])
129 data_c(0) = ocp.b(1);
130 for (index_t r = 0; r < nx; ++r)
131 for (index_t c = 0; c < nx; ++c)
132 data_c(0, r, 0) += ocp.A(0)(r, c) * ocp.b(0)(c, 0);
133 // r̃₀ = r₀ + S₀ x₀
134 data_rq(0).bottom_rows(nx) = ocp.q(N);
135 data_rq(0).top_rows(nu) = ocp.r(0);
136 for (index_t r = 0; r < nu; ++r)
137 for (index_t c = 0; c < nx; ++c)
138 data_rq(0, r, 0) += ocp.S(0)(r, c) * ocp.b(0)(c, 0);
139 for (index_t i = 1; i < N; ++i) {
140 data_H(i).top_left(nu, nu) = ocp.R(i);
141 batmat::linalg::copy(ocp.S(i).transposed(), data_H(i).bottom_left(nx, nu));
142 data_H(i).top_right(nu, nx) = ocp.S(i);
143 data_H(i).bottom_right(nx, nx) = ocp.Q(i);
144 data_F(i).left_cols(nu) = ocp.B(i);
145 data_F(i).right_cols(nx) = ocp.A(i);
146 data_G(i - 1).left_cols(nu) = ocp.D(i);
147 data_G(i - 1).right_cols(nx) = ocp.C(i);
148 data_lb(i - 1) = ocp.b_min(i);
149 data_ub(i - 1) = ocp.b_max(i);
150 data_c(i) = ocp.b(i + 1);
151 data_rq(i).bottom_rows(nx) = ocp.q(i);
152 data_rq(i).top_rows(nu) = ocp.r(i);
153 }
154}

156template <class T>
157CyqloneStorage<T> CyqloneStorage<T>::build(const LinearOCPStorage &ocp, index_t ny_0) {
158 const auto [N, nx, nu, ny, ny_N] = ocp.dim;
159 // Count the number of input constraints in the first stage
160 std::vector<bool> Ju0(ny);
161 const auto ny_0_ = count_constr_0(ocp, Ju0);
162 if (ny_0 < 0)
163 ny_0 = ny_0_;
164 else if (ny_0 == 0) {
165 ny_0 = 0;
166 Ju0.assign(ny, false);
167 } else
168 BATMAT_ASSERT(ny_0_ <= ny_0);
169 CyqloneStorage<T> res{.N_horiz = N,
170 .nx = nx,
171 .nu = nu,
172 .ny = ny,
173 .ny_0 = ny_0,
174 .ny_N = ny_N,
175 .Ju0 = std::move(Ju0)};
176 res.update_impl(ocp);
177 return res;
178}

181template <class T>
182void FanCyqloneStorage<T>::update_impl(const FanOCPStorage &focp) {
183 num_branches = focp.num_branches();
184 ny_0 = 0;
185 // Check that all OCPs in the fan have the same dimensions
186 for (index_t i = 0; i < num_branches; ++i) {
187 if (focp.ocps[i].dim.nu != focp.ocps[0].dim.nu) {
188 throw std::invalid_argument("All OCPs in the fan must have the same input dimension");
189 }
190 ny_0 += focp.ocps[i].D(0).rows;
191 }
192 nu = focp.ocps[0].dim.nu;
193 branches.resize(num_branches);
194 Bs.resize(num_branches);
195 R0 = matrix{{.rows = nu, .cols = nu}};
196 r0 = matrix{{.rows = nu, .cols = 1}};
197 D0 = matrix{{.rows = ny_0, .cols = nu}};
198 b_min_0 = matrix{{.rows = ny_0, .cols = 1}};
199 b_max_0 = matrix{{.rows = ny_0, .cols = 1}};
200 R0.set_constant(0);
201 r0.set_constant(0);
202 index_t ny_offset = 0;
203 for (index_t i = 0; i < num_branches; ++i) {
204 index_t ny_0i = focp.ocps[i].D(0).rows;
205 auto &cocp = branches[i] = CyqloneStorage<T>::build(focp.ocps[i], 0);
206 Bs[i] = matrix{{.rows = focp.ocps[i].dim.nx, .cols = nu}};
207 la::copy(cocp.data_F(0).left_cols(nu), Bs[i](0));
208 // Introduce an unused variable u(0) in each Cyqlone block
209 cocp.data_F(0).left_cols(nu).set_constant(0); // B(0) = 0
210 cocp.data_H(0).top_left(nu, nu).set_constant(0); // R(0) = I
211 cocp.data_H(0).top_left(nu, nu).set_diagonal(1); //
212 la::add(r0(0), cocp.data_rq(0).top_rows(nu));
213 la::add(R0(0), focp.ocps[i].R(0));
214
215 la::copy(focp.ocps[i].D(0), D0.middle_rows(ny_offset, ny_0i));
216 la::copy(focp.ocps[i].b_min(0), b_min_0.middle_rows(ny_offset, ny_0i));
217 la::copy(focp.ocps[i].b_max(0), b_max_0.middle_rows(ny_offset, ny_0i));
218 la::gemv_sub(focp.ocps[i].C(0), focp.ocps[i].b(0), b_min_0.middle_rows(ny_offset, ny_0i));
219 la::gemv_sub(focp.ocps[i].C(0), focp.ocps[i].b(0), b_max_0.middle_rows(ny_offset, ny_0i));
220
221 ny_offset += ny_0i;
222 cocp.data_rq(0).top_rows(nu).set_constant(0); // r(0) = 0
223 }
224}

239template <class T>
240auto FanCyqloneStorage<T>::reconstruct_solution(const FanOCPStorage &ocp,
241 std::span<const value_type> ux_compressed,
242 std::span<const value_type> y_compressed,
243 std::span<const value_type> λ_compressed) const
244 -> Solution {
245 Solution sol;
246 index_t offset_ux = 0;
247 index_t offset_y = 0;
248 index_t offset_λ = 0;
249 index_t ny_offset = 0;
250 sol.solutions.resize(num_branches);
251 for (index_t b = 0; b < num_branches; ++b) {
252 index_t ny_0i = ocp.ocps[b].D(0).rows;
253 const auto &cocp = branches[b];
254 sol.solutions[b].solution.resize(
255 static_cast<size_t>(cocp.N_horiz * (cocp.nu + cocp.nx) + cocp.nx));
256 sol.solutions[b].inequality_multipliers.resize(
257 static_cast<size_t>((cocp.N_horiz - 1) * (cocp.ny) + cocp.ny_N + ny_0i));
258 sol.solutions[b].equality_multipliers.resize(
259 static_cast<size_t>(cocp.N_horiz * cocp.nx + cocp.nx));
260 const index_t ux_size =
261 static_cast<std::size_t>((cocp.N_horiz - 1) * (cocp.nu + cocp.nx) + cocp.nx);
262 const index_t y_size = static_cast<std::size_t>((cocp.N_horiz - 1) * (cocp.ny) + cocp.ny_N);
263 const index_t λ_size = static_cast<std::size_t>(cocp.N_horiz * cocp.nx);
264 std::ranges::copy_n(ocp.ocps[b].b(0).data, cocp.nx,
265 sol.solutions[b].solution.begin()); // x0
266 std::ranges::copy(ux_compressed.subspan(0, cocp.nu),
267 sol.solutions[b].solution.begin() + cocp.nx); //u0
268 std::ranges::copy(ux_compressed.subspan(cocp.nu + offset_ux, ux_size),
269 sol.solutions[b].solution.begin() + cocp.nx +
270 cocp.nu); // x1u1..xN-1uN-1xN
271 std::ranges::copy(λ_compressed.subspan(offset_λ, λ_size),
272 sol.solutions[b].equality_multipliers.begin() + cocp.nx);
273 std::ranges::copy(y_compressed.subspan(ny_0 + offset_y, y_size),
274 sol.solutions[b].inequality_multipliers.begin() + ny_0i);
275 std::ranges::copy(y_compressed.subspan(ny_offset, ny_0i),
276 sol.solutions[b].inequality_multipliers.begin());
277 using MatrixView = guanaqo::MatrixView<value_type, index_t>;
278 MatrixView λ0{
279 {.data = sol.solutions[b].equality_multipliers.data(), .rows = cocp.nx, .cols = 1}},
280 λ1{{.data = sol.solutions[b].equality_multipliers.data() + cocp.nx,
281 .rows = cocp.nx,
282 .cols = 1}},
283 x0{{.data = sol.solutions[b].solution.data(), .rows = cocp.nx, .cols = 1}},
284 u0{{.data = sol.solutions[b].solution.data() + cocp.nx, .rows = cocp.nu, .cols = 1}};
285 λ0 = ocp.ocps[b].q(0);
286 guanaqo::blas::xsymv_L(value_type{1}, ocp.ocps[b].Q(0), x0, value_type{1}, λ0);
287 guanaqo::blas::xgemv_T(value_type{1}, ocp.ocps[b].S(0), u0, value_type{1}, λ0);
288 guanaqo::blas::xgemv_T(value_type{1}, ocp.ocps[b].A(0), λ1, value_type{1}, λ0);
289 if (ny_0i > 0) {
290 MatrixView y0{
291 {.data = sol.solutions[b].inequality_multipliers.data(), .rows = ny_0i, .cols = 1}};
292 guanaqo::blas::xgemv_T(value_type{1}, ocp.ocps[b].C(0), y0, value_type{1}, λ0);
293 }
294 offset_y += y_size;
295 offset_λ += λ_size;
296 offset_ux += ux_size;
297 ny_offset += ny_0i;
298 }
299
300 return sol;
301}
