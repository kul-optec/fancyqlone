45 void factor() {
46 for (index_t i = num_blocks; i-- > 0;) {
47 auto Mi = bla::tril(L.batch(i));
48
49 if (i + 1 == num_blocks) {
50 bla::potrf(Mi); // L(n-1) = chol(M(n-1))
51 // small_potrf_left(Mi); // L(n-1) = chol(M(n-1))
52 } else {
53 // bla::syrk_sub(U.batch(i), Mi);

56 }
57 if (i > 0)
