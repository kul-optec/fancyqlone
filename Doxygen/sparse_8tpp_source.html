16 std::span<const value_type> Σ) const
17 -> SparseMatrix {
18 BATMAT_ASSERT(is_pow_2(p));
19 using enum batmat::linalg::MatrixStructure;
20 using std::sqrt;
21 const index_t nux = nu + nx, nuxx = nux + nx;
22 // stride between stages in the same interval
23 const index_t stage_stride = nuxx;
24 // stride between intervals in the thread partitioning
25 const index_t interval_stride = nuxx * n - nx;
26 // stride between vector lanes (largest stride)
27 const index_t vector_stride = p * interval_stride;
28 // total matrix size
29 const index_t nn = ceil_N() * nuxx;
30 // first index of the CR block
31 const index_t sλ = nn - (nx * p * v);
32
33 SparseMatrixBuilder mat{.rows = nn, .cols = nn, .symmetry = Symmetry::Unsymmetric};
34
35 // Add the ALM penalty terms to the cost Hessians
36 batmat::matrix::Matrix<value_type, index_t> H{{.depth = N_horiz, .rows = nux, .cols = nux}};
37 const auto nyM = std::max(ny, ny_0 + ny_N);
38 batmat::matrix::Matrix<value_type, index_t> DC{{.depth = N_horiz, .rows = nyM, .cols = nux}};
39 auto R = H.top_left(nu, nu);
40 auto Q = H.bottom_right(nx, nx);
41 auto Sᵀ = H.bottom_left(nx, nu);
42 for (index_t j = 0; j < N_horiz; ++j) {
43 H(j) = ocp.data_H(j);
44 if (j > 0) {
45 DC.top_rows(ny)(j) = ocp.data_G(j - 1);
46 for (index_t r = 0; r < ny; ++r)
47 for (index_t c = 0; c < nux; ++c)
48 DC(j, r, c) *= sqrt(Σ[ny_0 + (j - 1) * ny + r]);
49 guanaqo::blas::xsyrk_LT(value_type{1}, DC(j), value_type{1}, H(j));
50 } else {
51 DC.top_rows(ny_0 + ny_N)(j) = ocp.data_G0N(0);
52 for (index_t r = 0; r < ny_0; ++r)
53 for (index_t c = 0; c < nu; ++c)
54 DC(0, r, c) *= sqrt(Σ[r]);
55 for (index_t r = 0; r < ny_N; ++r)
56 for (index_t c = 0; c < nx; ++c)
57 DC(0, ny_0 + r, nu + c) *= sqrt(Σ[ny_0 + (N_horiz - 1) * ny + r]);
58 guanaqo::blas::xsyrk_LT(value_type{1}, DC(j), value_type{1}, H(j));
59 }
60 }
61
62 // Populate the sparse matrix
63 for (index_t l = 0; l < v; ++l) { // vector lane
64 for (index_t c = 0; c < p; ++c) {
65 const auto biA = c + l * p;
66 const auto biI = sub_wrap_ceil_P(biA, 1);
67 const auto sλA = sλ + nx * get_linear_batch_offset(biA);
68 const auto sλI = sλ + nx * get_linear_batch_offset(biI);
69 // TODO: handle case if lev > or >= lp()
70 for (index_t i = 0; i < n; ++i) {
71 const index_t j = stage_index(c, i, l);
72 // index of current diagonal block
73 index_t s = l * vector_stride + c * interval_stride + stage_stride * i;
74 // Padding
75 if (j >= N_horiz) {
76 mat.add_diag(s, s, 1, nux); // H = I
77 i + 1 == n ? mat.add_diag(sλI, s + nu, -1, nx) // E = -I
78 : mat.add_diag(s + nux, s + nu, -1, nx); // E = -I
79 continue;
80 }
81 // H
82 mat.add(s, s, R(j), 1, LowerTriangular);
83 mat.add(s + nu, s + nu, Q(j), 1, LowerTriangular);
84 mat.add(s + nu, s, Sᵀ(j));
85 // beginning of subinterval: coupling constraints at the bottom (row sλA)
86 if (i == 0) {
87 mat.add(sλA, s, ocp.data_F(j).left_cols(nu)); // B(j)
88 j > 0 ? mat.add(sλA, s + nu, ocp.data_F(j).right_cols(nx)) // A(j)
89 : void(); // A(0) = 0
90 } else {
91 mat.add(s, s - nx, ocp.data_F(j).left_cols(nu).transposed()); // B(j)
92 mat.add(s + nu, s - nx, ocp.data_F(j).right_cols(nx).transposed()); // A(j)
93 }
94 // end of subinterval: coupling constraints at the bottom (row sλI)
95 i + 1 == n ? mat.add_diag(sλI, s + nu, -1, nx) // E = -I
96 : mat.add_diag(s + nux, s + nu, -1, nx); // E = -I
97 }
98 }
99 }
100 return std::move(mat).build();
101}

165 value_type scale_b) const
166 -> std::vector<T> {
167 BATMAT_ASSERT(is_pow_2(p));
168 const index_t nux = nu + nx, nuxx = nux + nx;
169 // stride between stages in the same interval
170 const index_t stage_stride = nuxx;
171 // stride between intervals in the thread partitioning
172 const index_t interval_stride = nuxx * n - nx;

174 const index_t vector_stride = p * interval_stride;

176 const index_t nn = ceil_N() * nuxx;

178 const index_t sλ = nn - (nx * p * v);

181 std::ranges::fill(rhs, std::numeric_limits<value_type>::quiet_NaN());

183 for (index_t l = 0; l < v; ++l) {
184 for (index_t t = 0; t < p; ++t) {
185 const index_t di0 = t * n;
186 for (index_t i = 0; i < n; ++i) {

188 index_t s = l * vector_stride + t * interval_stride + stage_stride * i;
189 if (i > 0)
190 for (index_t c = 0; c < nx; ++c)
191 rhs[s - nx + c] = scale_b * b.batch(di)(l)(c, 0);
192 for (index_t c = 0; c < nux; ++c)
193 rhs[s + c] = scale_rq * rq.batch(di)(l)(c, 0);
194 }
195 }
196 }
197 index_t s = sλ;
198 const auto cyclic_block = [&](index_t i) {
199 const index_t t = i % p, l = i / p;
200 const index_t di = t * n;
201 for (index_t c = 0; c < nx; ++c)
202 rhs[s + c] = scale_b * b.batch(di)(l)(c, 0);
203 s += nx;
204 };
205 for (index_t l = 0; l < lp(); ++l) {
206 index_t offset = 1 << l;
207 index_t stride = offset << 1;
208 for (index_t i = offset; i < v * p; i += stride)
209 cyclic_block(i);
210 }
211 for (index_t i = 0; i < v * p; i += p) {
212 cyclic_block(i);
213 }
214 return rhs;

218auto CyqloneSolver<VL, T, DefaultOrder, Ctx>::build_sparse_factor() const -> SparseMatrix {
219 BATMAT_ASSERT(is_pow_2(p));
220 using enum batmat::linalg::MatrixStructure;
221 const index_t nux = nu + nx, nuxx = nux + nx;
222 // stride between stages in the same interval
223 const index_t stage_stride = nuxx;
224 // stride between intervals in the thread partitioning
225 const index_t interval_stride = nuxx * n - nx;
226 // stride between vector lanes (largest stride)
227 const index_t vector_stride = p * interval_stride;
228 // total matrix size
229 const index_t nn = ceil_N() * nuxx;
230 // first index of the CR block
231 const index_t sλ = nn - (nx * p * v);
232
233 SparseMatrixBuilder mat{.rows = nn, .cols = nn, .symmetry = Symmetry::Unsymmetric};
234
235 // Compute blocks that are not computed by the Cyqlone factorization (because they are not
236 // needed for the solve)
237 matrix<> LA{{.depth = v * p, .rows = nx, .cols = n * nx}};
238 matrix<> invQᵀ{{.depth = v * p, .rows = nx, .cols = n * nx}};
239 matrix<> LBA{{.depth = v * p, .rows = nu + nx, .cols = (n - 1) * nx}};
240 for (index_t t = 0; t < p; ++t) {
241 const index_t di0 = t * n; // data batch index
242 auto RSQ = riccati_LH.batch(t);
243 auto LBAt = LBA.batch(t);
244 auto invQᵀt = invQᵀ.batch(t);
245 auto Â = riccati_LAB.batch(t).left_cols(n * nx);
246 auto LAt = LA.batch(t);
247 for (index_t i = 0; i < n; ++i) {
248 const auto di = di0 + i;
249 auto RSQi = RSQ.middle_cols(i * nux, nux);
250 auto Qi = tril(RSQi.bottom_right(nx, nx));
251 auto Qi_inv = triu(invQᵀt.middle_cols(i * nx, nx));
252 auto Âi = Â.middle_cols(i * nx, nx);
253 auto LAi = LAt.middle_cols(i * nx, nx);
254
255 copy(Âi, LAi);
256 if (i + 1 < n) // Final block already inverted
257 trtri(Qi, Qi_inv.transposed());
258 else
259 copy(triu(RSQi.block(nu - 1, nu, nx, nx)), Qi_inv);
260 if (i + 1 < n) // Final block is already Â LQ⁻ᵀ
261 trsm(LAi, Qi.transposed());
262 if (i > 0) {
263 auto LBAi = LBAt.middle_cols((i - 1) * nx, nx);
264 auto RSQ_prev = RSQ.middle_cols((i - 1) * nux, nux);
265 auto Q_prev = tril(RSQ_prev.bottom_right(nx, nx));
266 auto BA = data_F.batch(di);
267 copy(BA.transposed(), LBAi);
268 trmm(LBAi, Q_prev);
269 }
270 }
271 }
272
273 // Populate the sparse matrix
274 for (index_t l = 0; l < v; ++l) {
275 for (index_t t = 0; t < p; ++t) {
276 const auto biA = t + l * p;
277 const auto biI = sub_wrap_ceil_P(biA, 1);
278 const auto sλA = sλ + nx * get_linear_batch_offset(biA);
279 const auto sλI = sλ + nx * get_linear_batch_offset(biI);
280 auto B̂t = riccati_LAB.batch(t).right_cols(n * nu);
281 auto LHt = riccati_LH.batch(t);
282 auto LBAt = LBA.batch(t);
283 auto LAt = LA.batch(t);
284 auto invQᵀt = invQᵀ.batch(t);
285 for (index_t i = 0; i < n; ++i) {
286 index_t s = l * vector_stride + t * interval_stride + i * stage_stride;
287 auto B̂i = B̂t.middle_cols(i * nu, nu);
288 auto LHi = LHt.middle_cols(i * nux, nux);
289 auto iQiᵀ = invQᵀt.middle_cols(i * nx, nx);
290 auto AiQiᵀ = LAt.middle_cols(i * nx, nx);
291 if (i > 0) {
292 auto LBAi = LBAt.middle_cols((i - 1) * nx, nx);
293 auto iQᵀprev = invQᵀt.middle_cols((i - 1) * nx, nx);
294 auto LA_prev = LAt.middle_cols((i - 1) * nx, nx);
295 mat.add(s - nx, s - nx, iQᵀprev(l), -1, UpperTriangular);
296 mat.add(s, s - nx, LBAi(l));
297 mat.add(sλA, s - nx, LA_prev(l));
298 }
299 mat.add(s, s, LHi(l), 1, LowerTriangular);
300 mat.add(sλA, s, B̂i(l));
301 if (i + 1 < n)
302 mat.add(s + nux, s + nu, iQiᵀ(l), -1, UpperTriangular);
303 else
304 mat.add(sλI, s + nu, iQiᵀ(l), -1, UpperTriangular);
305 mat.add(sλA, s + nu, AiQiᵀ(l));
306 }
307 }
308 }
309 index_t s = sλ;
310 const auto cyclic_block = [&](index_t i, index_t offset) {
311 const index_t sY = sλ + nx * get_linear_batch_offset(i + offset);
312 const index_t sU = sλ + nx * get_linear_batch_offset(i - offset);
313 const index_t t = i % p, l = i / p;
314 mat.add(s, s, tricyqle.cr_L.batch(t)(l), 1, LowerTriangular);
315 if (i + offset < v * p)
316 mat.add(sY, s, tricyqle.cr_Y.batch(t)(l));
317 mat.add(sU, s, tricyqle.cr_U.batch(t)(l));
318 s += nx;
319 };
320 const auto cyclic_block_final = [&](index_t i, index_t offset) {
321 const index_t sY = sλ + nx * get_linear_batch_offset(i + offset);
322 const index_t t = i % p, l = i / p;
323 mat.add(s, s, tricyqle.pcr_L.batch(0)(l), 1, LowerTriangular);
324 if (i + offset < v * p)
325 mat.add(sY, s, tricyqle.cr_Y.batch(t)(l));
326 s += nx;
327 };
328 for (index_t l = 0; l < lp(); ++l) {
329 index_t offset = 1 << l;
330 index_t stride = offset << 1;
331 for (index_t i = offset; i < v * p; i += stride)
332 cyclic_block(i, offset);
333 }
334 for (index_t i = 0; i < v * p; i += p)
335 cyclic_block_final(i, p);
336 return std::move(mat).build();
337}
