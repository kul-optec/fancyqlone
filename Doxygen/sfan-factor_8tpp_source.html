109template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
110void SerialFanCyqloneSolver<VL, T, DefaultOrder, Ctx>::solve_reverse_λ0(index_t branch,
111 FanStorage &ux,
112 FanStorage &λ) const {
113 GUANAQO_TRACE_REGION("fan solve reverse λ0", 0);
114 // Compute the contribution of u(0) to λ(0) for each Cyqlone block
115 index_t nx = solver_storages[branch].cyqlone_solver.nx;
116 auto λ_last = solver_storages[branch].λ_last;
117 la::unpack(λ.branches[branch].batch(0), λ_last);
118 la::gemv_add(solver_storages[branch].LB.batch(0).transposed(), ux.first.batch(0).top_rows(nu),
119 λ_last.batch(0));
120 // Reverse solve of the final block-tridiagonal system for each Cyqlone block
121 solver_storages[branch].tridiagonal_solver.solve_reverse(λ_last);
122 la::pack(λ_last, λ.branches[branch].batch(0));
123 la::copy(λ_last.batch(0), λ.first.batch(0).middle_rows(branch * nx, nx));
124};

181// §3.2 Cyclic reduction of block-tridiagonal linear systems
182//
183// The reverse solve routines below closely follow the structure of the corresponding factorization
184// and forward solve routines, but in reverse order. An iterative approach is used instead of
185// recursion. Note that the evaluation of λ(0) is performed during the forward solve step.
186// Depending on the problem size, either a parallel or serial version of the CR solve is used.
187
188template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>

190 FanStorage &λ) {

192
193 if (ctx.is_master()) {
194 solve_reverse_u0(ux);
195 for (index_t b = 0; b < num_branches; ++b) {
196 solve_reverse_λ0(b, ux, λ);

201 solver_storages[b].cyqlone_solver.solve_reverse(ctx, ux.branches[b], λ.branches[b]);
