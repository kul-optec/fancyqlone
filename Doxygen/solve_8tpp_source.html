19 Context &ctx, [[maybe_unused]] const var_vec_t &x, const var_vec_t &grad, const var_vec_t &Mᵀλ,
20 const var_vec_t &Aᵀŷ, const eq_constr_vec_t &Mxb, real_t S,
21 [[maybe_unused]] const ineq_constr_vec_t &Σ,
22 const active_set_t &J, //
23 var_vec_t &d, var_vec_t &ξ, ineq_constr_vec_t &Ad, eq_constr_vec_t &Δλ, var_vec_t &MᵀΔλ) {
24 const auto rhs = [&](auto di, auto gi, auto Mᵀλi, auto Aᵀŷi, auto Mxbi, auto Δλi) {
25 linalg::axpy<0>(di, {-1, -1, -1}, gi, Mᵀλi, Aᵀŷi);
26 batmat::linalg::copy(Mxbi, Δλi);
27 };
28 ocp.foreach_batch(ctx, rhs, d, grad, Mᵀλ, Aᵀŷ, Mxb, Δλ);
29 if (settings.print_residuals)
30 print_solve_rhs_norms(ctx, d, Δλ, grad, Mᵀλ, Aᵀŷ);
31 if (reset_factorization) {
32 auto t = get_timed(&Timings::factor);
33 ocp.factor_solve(ctx, S, 0, J, d, Δλ); // (d, Δλ) ← L⁻¹ (d, Δλ)
34 if (ctx.is_master()) {
35 // No synchronization needed here, barriers in factor_solve and solve_reverse
36 reset_factorization = false;
37 update_pending = false;
38 num_updates = 0;
39 ++stats.num_factor;
40 }
41 } else if (update_pending) {
42 auto t = get_timed(&Timings::update_factorization);
43 ocp.update_solve(ctx, ΔΣ, d, Δλ); // (d, Δλ) ← L⁻¹ (d, Δλ)
44 if (ctx.is_master()) {
45 // No synchronization needed here, barriers in update_solve and solve_reverse
46 update_pending = false;
47 }
48 } else { // (d, Δλ) ← L⁻¹ (d, Δλ)
49 auto t = get_timed(&Timings::solve);
50 ocp.solve_forward(ctx, d, Δλ);
51 }
52
53 { // (d, Δλ) ← L⁻ᵀ (d, Δλ) and MᵀΔλ ← Mᵀ Δλ
54 auto t = get_timed(&Timings::solve);
55 ocp.solve_reverse_mul(ctx, d, Δλ, MᵀΔλ);
56 }
57 // ocp.print_grad(ctx, d, "d (after reverse solve)");
58 // ocp.print_grad(ctx, Δλ, "Δλ (after reverse solve)");
59
60 { // Ad ← A d4
61 auto t = get_timed(&Timings::solve_A);
62 mat_vec_A(ctx, d, Ad);
63 }
64 { // ξ ← Q d + S⁻¹ d
65
66 auto t = get_timed(&Timings::solve_grad);
67 ocp.cost_gradient(ctx, d, 1 / S, d, 0, ξ);
68 }
69 if (settings.print_residuals)
70 print_solve_resid_norms(ctx, x, d, grad, ξ, Mᵀλ, Aᵀŷ, MᵀΔλ, Ad, J);
71}
