solve-ocp.cpp
Example demonstrating how to solve an optimal control problem using Cyqlone.

This example generates a random optimal control problem (without inequality constraints), and solves the resulting KKT system using Cyqlone.

20// Generate a random linear-quadratic OCP.
21fancyqlone::LinearOCPStorage init_ocp(index_t N, index_t nx, index_t nu) {
22 using batmat::linalg::tril;
23 using batmat::linalg::triu;
24
25 std::mt19937 rng{12345};
26 std::uniform_real_distribution<real_t> uni_dist{-1.2, 1.2};
27 auto uni_gen = [&] { return uni_dist(rng); };
28
29 fancyqlone::LinearOCPStorage ocp{.dim{.N_horiz = N, .nx = nx, .nu = nu, .ny = 0}};
30 auto diag = 2 * static_cast<real_t>(nx + nu);
31 ocp.b(0).generate(uni_gen); // initial state x(0)
32 for (index_t j = 0; j < N; ++j) {
33 ocp.A(j).generate(uni_gen); // Dynamics Jacobian df/dx
34 ocp.B(j).generate(uni_gen); // Dynamics Jacobian df/du
35 ocp.b(j + 1).generate(uni_gen); // Dynamics constant term f(j)
36 ocp.Q(j).generate(uni_gen); // Cost Hessian d²ℓ/dx²
37 ocp.R(j).generate(uni_gen); // Cost Hessian d²ℓ/du²
38 ocp.S(j).generate(uni_gen); // Cost Hessian d²ℓ/dxdu
39 ocp.q(j).generate(uni_gen); // Cost gradient dℓ/dx
40 ocp.r(j).generate(uni_gen); // Cost gradient dℓ/du
41 ocp.Q(j).add_to_diagonal(diag); // ensure positive definiteness
42 ocp.R(j).add_to_diagonal(diag);
43 copy(tril(ocp.H(j)).transposed(), triu(ocp.H(j))); // ensure symmetry of cost Hessian
44 }
45 ocp.Q(N).generate(uni_gen); // Terminal cost Hessian d²ℓ_N(x)/dx²
46 ocp.q(N).generate(uni_gen);
47 ocp.Q(N).add_to_diagonal(diag);
48 copy(tril(ocp.Q(N)).transposed(), triu(ocp.Q(N)));
49 return ocp;
50}

52// Example of solving a linear-quadratic OCP using the CyqloneSolver.
53int main(int argc, char *argv[]) try {
54 // Problem dimensions and parameters
55 const index_t p = argc > 1 ? std::stoi(argv[1]) : 8; // Number of processors/threads
56 const index_t N = argc > 2 ? std::stoi(argv[2]) : 128; // Horizon length
57 const index_t nx = argc > 3 ? std::stoi(argv[3]) : 10; // State dimension
58 const index_t nu = argc > 4 ? std::stoi(argv[4]) : 5; // Control dimension
59 BATMAT_ASSERT(v == 1 || fancyqlone::is_pow_2(p));
60
61 // Initialize a random linear-quadratic OCP
62 auto ocp = init_ocp(N, nx, nu);
63 // Eliminate the initial state x0 and merge the first and last stages for Cyqlone
64 auto cocp = fancyqlone::CyqloneStorage<real_t>::build(ocp);
65
66 // Create a Cyqlone solver for the OCP
67 fancyqlone::CyqloneSolver<v, real_t> solver =
68 fancyqlone::CyqloneSolver<v, real_t>::build(cocp, p);
69 // Optionally configure the solver parameters
70 auto params = solver.get_tricyqle_params();
71 params.solve_method = fancyqlone::SolveMethod::PCR; // Use the PCR solver instead of PCG
72 solver.update_tricyqle_params(params);
73 std::println("{}", solver.get_params_string());
74
75 // Initialize copies of the OCP data vectors in compact storage format
76 const auto rq = solver.initialize_gradient(cocp); // gradient vectors r and q
77 const auto b = solver.initialize_rhs(cocp); // right-hand side vector b of dynamics constraints
78 // Initialize the right-hand side of the KKT system (will be overwritten by the solution)
79 auto ux = solver.initialize_variables(), λ = solver.initialize_dynamics_constraints();
80 fancyqlone::linalg::negate(rq, ux); // right-hand side of the KKT system is negative gradient
81 fancyqlone::linalg::copy(b, λ);
82 const real_t γ = 1e99; // Primal regularization
83
84 // Call the solver in parallel, passing a lambda that is executed by each thread
85 auto pctx = solver.create_parallel_context();
86 auto t0 = std::chrono::high_resolution_clock::now();
87 pctx->run([&](fancyqlone::CyqloneSolver<v, real_t>::Context &ctx) {
88 solver.factor_solve(ctx, γ, 0, {}, ux, λ); // Cholesky factorization & forward substitution
89 solver.solve_reverse(ctx, ux, λ); // Backward substitution
90 }); // blocks until all threads have joined
91 auto t1 = std::chrono::high_resolution_clock::now();
92 std::println("Time: {:.3f} ms", std::chrono::duration<double, std::milli>(t1 - t0).count());
93
94 // Compute the residuals (in compact storage and without x0, also in parallel)
95 auto Mxb = solver.initialize_dynamics_constraints(); // primal equality constraint residual
96 auto grad = solver.initialize_variables(); // gradient of the Lagrangian
97 pctx->run([&](fancyqlone::CyqloneSolver<v, real_t>::Context &ctx) {
98 solver.residual_dynamics_constr(ctx, ux, b, Mxb); // Mx + b
99 solver.transposed_dynamics_constr(ctx, λ, grad); // Mᵀλ
100 solver.cost_gradient(ctx, ux, 1, rq, 1, grad); // + Qx + q
101 });
102
103 // Print the residuals
104 using fancyqlone::linalg::norm_inf;
105 std::println("Eq. residual (compact): {:.17e}", norm_inf(Mxb));
106 std::println("Lagr. stationarity (compact): {:.17e}", norm_inf(grad));
107
108 // Reconstruct the full solution (as a flat vector, including x0)
109 std::vector ux_unpacked = solver.unpack_variables(ux); // convert compact to linear storage
110 std::vector λ_unpacked = solver.unpack_dynamics(λ);
111 auto sol = cocp.reconstruct_solution(ocp, ux_unpacked, {}, λ_unpacked);
112
113 // Check the residuals of the original OCP
114 auto resid = ocp.compute_kkt_error(sol);
115 std::println("Eq. residual (full): {:.17e}", resid.equality_residual);
116 std::println("Lagr. stationarity (full): {:.17e}", resid.stationarity);
117
118#if CYQLONE_WITH_MATIO
119 // Export the original OCP and the solution as a .mat file
120 std::filesystem::path filename = "ocp_solution.mat";
121 auto matfile = fancyqlone::create_mat(filename);
122 fancyqlone::add_to_mat(matfile.get(), "ocp", ocp);
123 fancyqlone::add_to_mat(matfile.get(), "sol_x", sol.solution);
124 fancyqlone::add_to_mat(matfile.get(), "sol_λ", sol.equality_multipliers);
125 std::cout << "Saved OCP and solution to " << filename << "\n";
126#endif
127} catch (std::exception &e) {
128 std::cerr << "Error: " << e.what() << "\n";
129 return 1;
130}
