47/**
48 * Fairly vanilla combining tree barrier. It is inspired by GCC 15.2's __tree_barrier, with some
49 * important API differences:
50 * - Every thread has a unique thread ID in [0, expected-1]. This eliminates the need for hashing
51 * the pthread thread IDs and for the inner search loop to find free slots in the tree.
52 * - Wait tries to spin for a given number of iterations before falling back to a futex-based
53 * atomic wait.
54 * - The barrier phase is exposed to the user.
55 * - Custom completion functions can be provided at arrival time.
56 * - Reductions and broadcasts on small values are supported.
57 * @ingroup topic-parallelization
58 */
59template <typename CompletionFn = EmptyCompletion, class PhaseType = uint32_t>

73 BarrierPhase get() const noexcept { return phase; }

129 /// Atomic counters for each level of the combining tree. Aligned to avoid false sharing.
130 /// @todo figure out why the libstdc++ implementation does not reuse tickets across levels
131 /// (as is done with the storage in the reduction variant below).
132 struct alignas(cache_line_size) State {
133 using atomic_byte = std::atomic<unsigned char>;
134 using atomic_word = std::atomic<uint32_t>;
135 static constexpr bool only_word_lock_free =
136 atomic_word::is_always_lock_free && !atomic_byte::is_always_lock_free;
137 // Make the ticket size as small as possible while still being lock-free.
138 // TODO: this may be overkill and/or even unnecessary if we reuse tickets.
139 using ticket_t = std::conditional_t<only_word_lock_free, atomic_word, atomic_byte>;
140 static constexpr size_t num_levels = cache_line_size / sizeof(ticket_t);
141 std::array<ticket_t, num_levels> tickets{};
142 };
143 using ticket_value_type = typename State::ticket_t::value_type;

176 /// Combining tree arrival. The last thread arriving at a certain ticket (counter) moves on to
177 /// the next level of the tree. When reaching the root, it returns true. The number of tickets
178 /// halves at each level, with at most two threads per ticket.
179 bool arrive_impl(BarrierPhase old_phase, uint32_t thread_id) {
180 static constexpr auto acq_rel = std::memory_order_acq_rel;
181 const auto first_of_one = static_cast<ticket_value_type>(old_phase),
182 second_of_two = static_cast<ticket_value_type>(2 * first_of_one + 1);
183
184 uint32_t level_size = expected; // Total sum in this level of the tree
185 for (size_t level = 0;; ++level) {
186 if (level_size <= 1)
187 return true;
188 BATMAT_ASSUME(level < 32);
189 thread_id >>= 1;
190 auto &ticket = state[thread_id].tickets[level];
191 const uint32_t end_node = (level_size + 1) >> 1; // Two threads per node
192 const bool last_odd = thread_id + 1 == end_node && (level_size & 1) == 1;
193 const auto target = last_odd ? first_of_one : second_of_two;
194 const auto old_value = ticket.fetch_add(1, acq_rel);
195 if (old_value != target)
196 return false;
197 level_size = end_node;
198 }
199 }

201 /// Fused implementation of the combining tree arrival and a reduction operation. The last
202 /// thread arriving at a certain ticket (counter) moves on to the next level of the tree. When
203 /// it does so, it reads the value written by the other thread that arrived at the same ticket,
204 /// applies the reduction function, and writes the result to be used in the next level. When
205 /// reaching the root, it stores the final value and returns true. Note that the left and right
206 /// arguments to the reduction function are determined by the thread IDs, regardless of the
207 /// order in which threads arrive. In other words, for a given number of threads, the order of
208 /// the reduction operations is fully deterministic.
209 template <class T, class F>
210 std::optional<T> arrive_impl(BarrierPhase old_phase, uint32_t thread_id, T value, F reduce) {
211 static constexpr auto acq_rel = std::memory_order_acq_rel;
212 const auto first_of_one = static_cast<ticket_value_type>(old_phase),
213 second_of_two = static_cast<ticket_value_type>(2 * first_of_one + 1);
214
215 // Diagram of the storage used at each level (for expected = 4):
216 // l=0 l=1 l=2
217 // storage[0] t=0 1 3 10
218 // storage[1] t=1 2
219 // storage[2] t=2 3 7
220 // storage[3] t=3 4
221
222 uint32_t level_size = expected; // Total sum in this level of the tree
223 for (size_t level = 0;; ++level) {
224 if (level_size <= 1) {
225 return value;
226 }
227 BATMAT_ASSUME(level < 32);
228 auto offset = size_t{1} << level;
229 auto write = thread_id << level;
230 storage[write].store(value);
231 thread_id >>= 1;
232 auto &ticket = state[thread_id].tickets[level];
233 const uint32_t end_node = (level_size + 1) >> 1; // Two threads per node
234 const bool last_odd = thread_id + 1 == end_node && (level_size & 1) == 1;
235 const auto target = last_odd ? first_of_one : second_of_two;
236 const auto old_value = ticket.fetch_add(1, acq_rel);
237 if (old_value != target)
238 return std::nullopt;
239 if (!last_odd)
240 value = reduce(storage[write & ~offset].template load<T>(),
241 storage[write | +offset].template load<T>());
242 level_size = end_node;
243 }
244 }

246 /// Generic implementation of arrive with custom completion function. The arrival function
247 /// should return true when the thread is the last to arrive at the root of the tree.
248 /// Returns a token that can be used to wait for the barrier to complete.
249 /// The custom completion function is called by the last thread arriving at the root, before
250 /// advancing the barrier phase and notifying all waiting threads.
251 template <class A, class C>
252 [[nodiscard]] arrival_token arrive_with_completion(uint32_t thread_id, A arrival,
253 C &&custom_completion) {
254 BATMAT_ASSUME(thread_id < expected);
255 const auto cur_phase = phase.load(std::memory_order_relaxed);
256#if CYQLONE_SANITY_CHECKS_BARRIER
257 sanity_check_arrival(thread_id, cur_phase);
258#endif
259 if (auto arrived = arrival(cur_phase, thread_id); arrived) {
260 if constexpr (std::is_same_v<decltype(arrived), bool>)
261 std::invoke(std::forward<C>(custom_completion));
262 else
263 std::invoke(std::forward<C>(custom_completion), std::move(*arrived));
264 auto next_phase = static_cast<BarrierPhase>(static_cast<PhaseType>(cur_phase) + 1);
265 phase.store(next_phase, std::memory_order_release);
266 atomic_notify_fence();
267 phase.notify_all();
268 }
269 return arrival_token{cur_phase};
270 }

272 template <class T>
273 void atomic_wait_ne_acq(const std::atomic<T> &a, T old) const {
274 if (a.load(std::memory_order_acquire) != old) [[likely]]
275 return;
276 // Spin before calling wait
277 for (auto spin = this->spin_count; spin-- > 0;)
278 if (a.load(std::memory_order_acquire) != old) [[unlikely]]
279 return;
280 a.wait(old, std::memory_order_acquire);
281 }
282
283 template <class T>
284 void atomic_wait_eq_acq(const std::atomic<T> &a, T target) const {
285 auto old = a.load(std::memory_order_acquire);
286 if (old == target) [[likely]]
287 return;
288 // Spin before calling wait
289 for (auto spin = this->spin_count; spin-- > 0;)
290 if (old = a.load(std::memory_order_acquire); old == target) [[unlikely]]
291 return;
292 do {
293 a.wait(old, std::memory_order_acquire);
294 old = a.load(std::memory_order_acquire);
295 } while (old != target);
296 }

309 /// Create a barrier with @p expected participating threads and a completion function that is
310 /// called by the last thread that arrives at each phase.
311 TreeBarrier(uint32_t expected, CompletionFn completion)
312 : expected(expected), completion(std::move(completion)),
313 phase(static_cast<BarrierPhase>(0)) {
314 BATMAT_ASSERT(expected <= max());
315 const size_t leaf_count = (expected + 1) >> 1;
316 state = std::make_unique<State[]>(leaf_count);
317 storage = std::make_unique<Storage[]>(expected);

359 /// Arrive at the barrier, recording the given line number for sanity checking to make sure
360 /// that all threads arrive from the same line or statement in the source code.
361 /// This is useful for debugging purposes to detect mismatched barrier calls, but should not
362 /// really be used otherwise. If @ref CYQLONE_SANITY_CHECKS_BARRIER is disabled, the line
363 /// number is ignored and this function is equivalent to @ref arrive(uint32_t).
364 /// Each thread should use a unique thread ID in [0, expected-1].
365 [[nodiscard]] arrival_token arrive(uint32_t thread_id, [[maybe_unused]] int line) {
366#if CYQLONE_SANITY_CHECKS_BARRIER
367 get_local_line(thread_id).store(static_cast<ticket_value_type>(line),
368 std::memory_order_relaxed);
369 return arrive_with_completion(thread_id, [this, line, thread_id] {
370 for (uint32_t i = 0; i < expected; ++i)
371 if (get_local_line(i).load(std::memory_order_relaxed) !=
372 static_cast<ticket_value_type>(line))
373 throw std::logic_error(std::format(
374 "Barrier sanity check failed: thread {} arrived at line {}, "
375 "but thread {} arrived at line {}",
376 i, get_local_line(i).load(std::memory_order_relaxed), thread_id, line));
377 completion();
378 });
379#else
380 return arrive(thread_id);
381#endif
382 }

393 /// Check if @ref wait() may block. If it returns false, the caller can call @ref wait()
394 /// and it will return immediately without spinning or sleeping. This is useful if the caller
395 /// has other non-critical work to do while waiting for other threads.
396 /// Users should still call @ref wait() before arriving again.
397 /// @note This function does not impose any memory ordering, so even when it returns false,
398 /// changes made before the arrival of other threads may not be visible yet. In contrast,
399 /// @ref wait() does ensure proper synchronization.
400 bool wait_may_block(const arrival_token &token) const noexcept {
401 return phase.load(std::memory_order_relaxed) == token.get();
402 }

404 /// Wait for the barrier to complete after an arrival, using the given token. Separating the
405 /// arrival and wait phases allows for overlapping computation with waiting, hiding the
406 /// synchronization latency.
407 /// Waiting on the same token multiple times is not allowed.
408 void wait(arrival_token &&token) const {
409 const auto old_phase = token.get();
410 // barring overflow, we have that current_phase >= old_phase
411 atomic_wait_ne_acq(phase, old_phase);
412 }

439 /// Combining tree reduction across all threads. Deterministic application order for a given
440 /// number of threads.
441 template <class T, class F, class CF>
442 [[nodiscard]] arrival_token_typed<T> arrive_reduce(uint32_t thread_id, T x, F reduce,
443 CF custom_completion) {
444#if CYQLONE_SANITY_CHECKS_BARRIER
445 auto hash = typeid(T).hash_code() ^ typeid(F).hash_code();
446 get_local_line(thread_id).store(static_cast<ticket_value_type>(hash),
447 std::memory_order_relaxed);
448 auto completion = [this, hash, thread_id, &custom_completion](T x) {
449 for (uint32_t i = 0; i < expected; ++i)
450 if (get_local_line(i).load(std::memory_order_relaxed) !=
451 static_cast<ticket_value_type>(hash))
452 throw std::logic_error(std::format(
453 "Barrier sanity check failed: different types in arrive_reduce: "
454 "thread {} has hash {}, but thread {} has hash {}",
455 i, get_local_line(i).load(std::memory_order_relaxed), thread_id, hash));
456 std::invoke(custom_completion, x);
457 };
458#else
459
460 auto &completion = custom_completion;
461
462#endif
463 auto arrival = [this, &reduce, &x](BarrierPhase cur_phase, uint32_t thread_id) {
464 return arrive_impl(cur_phase, thread_id, std::move(x), std::move(reduce));
465 };
466 return arrival_token_typed<T>{arrive_with_completion(thread_id, arrival, completion)};

487 /// Broadcast a value from the source thread to all other threads. All threads must call this
488 /// function with the same source thread ID.
489 template <class T>
490 [[nodiscard]] T broadcast(uint32_t thread_id, T &&x, uint32_t src = 0) {
491 if (thread_id == src)
492 storage[thread_id].store(std::forward<T>(x));
493 // TODO: in debug mode, we could have the other threads write some unused bit in the storage
494 // to detect if all calls used the same src.

496 broadcast_storage.store(storage[src].template load<T>());
497 };
498 wait(arrive_with_completion(thread_id, custom_completion));

516 /// Publish the value @p x to another thread, and prepare to receive the value from thread
517 /// @p src. Note that the @p src and @p thread_id arguments should define a complete
518 /// permutation, i.e. each thread sends its value to exactly one other thread.
519 /// @note This does not complete a barrier phase.
520 template <class T>
521 [[nodiscard]] shuffle_token<T> shuffle_start(uint32_t thread_id, T x, uint32_t src) {
522 assert(thread_id < expected);
523 assert(src < expected);
524 auto &ticket = get_shuffle_ticket(thread_id);
525 // If the ticket is odd, then the previous shuffle has not completed yet, so wait for it.
526 if (auto old_ticket = ticket.load(std::memory_order_acquire); old_ticket & 1) [[unlikely]]
527 atomic_wait_ne_acq(ticket, old_ticket);
528 storage[thread_id].store(std::move(x));
529 auto old_ticket = ticket.fetch_add(1, std::memory_order_release);
530 atomic_notify_fence();
531 ticket.notify_one();
532 return {src, old_ticket};
533 }

535 /// Wait from the value sent by @ref shuffle_start, applying a function to it.
536 /// Note that if the value contains any references to local variables on the sending thread,
537 /// they can safely be accessed in @p func, but they become dangling after @p func returns,
538 /// because then the sending thread may continue execution asynchronously.
539 template <class T, class F>
540 requires(std::is_void_v<std::invoke_result_t<F &&, T &&>>)
541 void shuffle_wait(uint32_t thread_id, shuffle_token<T> &&token, F &&func) {
542 auto &ticket = get_shuffle_ticket(thread_id);
543 auto &src_ticket = get_shuffle_ticket(token.src);
544 atomic_wait_eq_acq(src_ticket, static_cast<ticket_value_type>(token.ticket + 1));
545 std::invoke(std::forward<F>(func), std::move(storage[token.src].template load<T>()));
546 src_ticket.fetch_add(1, std::memory_order_release);
547 atomic_notify_fence();
548 src_ticket.notify_one();
549 atomic_wait_eq_acq(ticket, static_cast<ticket_value_type>(token.ticket + 2));
550 }

552 template <class T, class F>
553 requires(!std::is_void_v<std::invoke_result_t<F &&, T &&>>)
554 [[nodiscard]]
555 std::invoke_result_t<F &&, T &&> shuffle_wait(uint32_t thread_id, shuffle_token<T> &&token,
556 F &&func) {
557 auto &ticket = get_shuffle_ticket(thread_id);
558 auto &src_ticket = get_shuffle_ticket(token.src);
559 atomic_wait_eq_acq(src_ticket, static_cast<ticket_value_type>(token.ticket + 1));
560 std::invoke_result_t<F &&, T &&> ret =
561 std::invoke(std::forward<F>(func), std::move(storage[token.src].template load<T>()));
562 src_ticket.fetch_add(1, std::memory_order_release);
563 atomic_notify_fence();
564 src_ticket.notify_one();
565 atomic_wait_eq_acq(ticket, static_cast<ticket_value_type>(token.ticket + 2));
566 return ret;
567 }

569 /// @warning Any pointers or references in the return value should be assumed to be dangling.
570 /// Use the variant that takes a function argument to access them safely.
571 template <class T>
572 T shuffle_wait(uint32_t thread_id, shuffle_token<T> &&token) {
573 (void)thread_id;
574 auto &src_ticket = get_shuffle_ticket(token.src);
575 atomic_wait_eq_acq(src_ticket, static_cast<ticket_value_type>(token.ticket + 1));
576 T ret = std::move(storage[token.src].template load<T>());
577 src_ticket.fetch_add(1, std::memory_order_release);
578 atomic_notify_fence();
579 src_ticket.notify_one();
580 // No need to wait for the value we sent to be consumed by the receiving thread, since it
581 // cannot contain any references to our local variables. The next shuffle_start will wait
582 // before overwriting the storage.
583 return ret;
584 }
