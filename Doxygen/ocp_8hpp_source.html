16/// Dimensions of an optimal control problem.

18struct OCPDim {

24 friend constexpr bool operator==(OCPDim, OCPDim) = default;

28/// Storage for a linear-quadratic OCP of the form
29/// ~~~
30/// ₙ₋₁
31/// minimize ∑ [½ uᵢᵀ Rᵢ uᵢ + uᵢᵀ S xᵢ + ½ xᵢᵀ Qᵢ xᵢ + rᵢᵀuᵢ + qᵢᵀxᵢ] + ½ xₙᵀ Qₙ xₙ + qₙᵀ xₙ
32/// ⁱ⁼⁰
33/// s.t. x₀ = b₀
34/// xᵢ₊₁ = Aᵢ xᵢ + Bᵢ uᵢ + bᵢ₊₁
35/// lᵢ ≤ Cᵢ xᵢ + Dᵢ uᵢ ≤ uᵢ
36/// lₙ ≤ Cₙ xₙ ≤ uₙ
37/// ~~~
38/// @ingroup topic-ocp-formulations
39struct LinearOCPStorage {
40 OCPDim dim{};
41 /// Create a single contiguous storage for all problem data.
42 /// ~~~
43 /// Storage layout: size offset
44 /// N × [ Q Sᵀ] = H (nx+nu)² 0
45 /// [ S R ]
46 /// 1 × [ Q ] nx² N (nx+nu)²
47 /// N × [ C D ] ny(nx+nu) N (nx+nu)² + nx²
48 /// 1 × [ C ] ny_N nx N (nx+nu+ny)(nx+nu) + nx²
49 /// N × [ A B ] nx(nx+nu) N (nx+nu+ny)(nx+nu) + (nx+ny_N)nx
50 /// N × [ q r ] nx+nu N (2nx+nu+ny)(nx+nu) + (nx+ny_N)nx
51 /// 1 × [ q ] nx N (2nx+nu+ny+1)(nx+nu) + (nx+ny_N)nx
52 /// N+1 × [ b ] nx N (2nx+nu+ny+1)(nx+nu) + (nx+ny_N+1)nx
53 /// N × [ l ] ny N (2nx+nu+ny+1)(nx+nu) + (nx+ny_N+2+N)nx
54 /// 1 × [ l ] ny_N N (2nx+nu+ny+1)(nx+nu) + (nx+ny_N+2+N)nx + N ny
55 /// N × [ u ] ny N (2nx+nu+ny+1)(nx+nu) + (nx+ny_N+2+N)nx + N ny + ny_N
56 /// 1 × [ u ] ny_N N (2nx+nu+ny+1)(nx+nu) + (nx+ny_N+2+N)nx + N (2ny) + ny_N
57 /// N (2nx+nu+ny+1)(nx+nu) + (nx+ny_N+2+N)nx + N (2ny) + 2ny_N
58 /// ~~~
59 std::vector<real_t> storage = create_storage(dim);
60 static std::vector<real_t> create_storage(OCPDim dim) {
61 auto [N, nx, nu, ny, ny_N] = dim;
62 index_t size = N * (2 * nx + nu + ny + 1) * (nx + nu) + (nx + ny_N + 2 + N) * nx +
63 2 * N * ny + 2 * ny_N;
64 return std::vector<real_t>(size);
65 }
