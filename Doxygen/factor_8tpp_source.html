47template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
48template <bool Factor, bool Solve>
49void TricyqleSolver<VL, T, DefaultOrder, Ctx>::factor_solve_skip_first(Context &ctx, mut_view<> λ,
50 index_t stride) {
51 // When vectorization is enabled, the number of threads p must be a power of two.
52 // TODO: allow circular coupling for v=1 and non-power-of-two p, which requires wrapping of
53 // the indices in the CR code.
54 BATMAT_ASSERT(is_pow_2(p) || (v == 1 && !circular));
55 const index_t c = ctx.index;
56 // 17| for l = 0 ... log₂(P)-1
57 for (index_t l = 0; l < lp(); ++l) { // Recursion level of cyclic reduction
58 const auto c_ = cr_thread_assignment(l, c);
59 // 18| iU = c+1, iY = c+1-2^l
60 const auto iU = add_wrap_ceil_p(c_, 1), iY = sub_wrap_ceil_p(c_, (1 << l) - 1);
61 // 19| -- sync --
62 ctx.arrive_and_wait(); // Wait for L
63 // 20| if ν₂(iU) = l: U(iU) = K˂(iU) L(iU)⁻ᵀ
64 if (ν2p(iU) == l) {
65 if constexpr (Factor)
66 factor_U(l, iU);
67 if constexpr (Solve)
68 solve_u_forward(l, iU, λ, stride);
69 }
70 // 21| elif ν₂(iY) = l: Y(iY) = K˃(iY) L(iY)⁻ᵀ
71 else if (ν2p(iY) == l) {
72 if constexpr (Factor)
73 factor_Y(l, iY);
74 if constexpr (Solve)
75 solve_y_forward(l, iY, λ, work_cr, stride);
76 }
77 // 22| -- sync --
78 ctx.arrive_and_wait(); // Wait for U, Y
79 // 23| if ν₂(iU) = l: factor-L(l, iY)
80 if (ν2p(iU) == l) {
81 if constexpr (Factor)
82 factor_L(l, iY);
83 if constexpr (Solve)
84 solve_λ_forward(l, iY, λ, work_cr, stride);
85 }
86 // 24| elif ν₂(iY) = l: update-K(l, iY)
87 else if (ν2p(iY) == l) {
88 if constexpr (Factor)
89 update_K(l, iY);
90 }
91 }
92 // Factor or solve the last level using PCR or PCG
93 if constexpr (Factor) {
94 if (params.solve_method == SolveMethod::PCR) {
95 ctx.arrive_and_wait(); // wait for off-diagonal block
96 if (block_size >= params.parallel_factor_pcr_threshold && p > 1)
97 factor_pcr_parallel(ctx);
98 else if (ν2p(c + 1) + 1 == lp() || p == 1)
99 factor_pcr();
100 }
101 }
102 if constexpr (Solve) {
103 if (params.solve_method == SolveMethod::PCR) {
104 if constexpr (!Factor)
105 ctx.arrive_and_wait(); // wait for off-diagonal block TODO: necessary?
106 if (ν2p(c + 1) + 1 == lp() || p == 1)
107 solve_pcr(λ.batch(0), work_pcg.batch(0).left_cols(1));
108 } else if (params.solve_method == SolveMethod::StairPCG ||
109 params.solve_method == SolveMethod::JacobiPCG) {
110 ctx.arrive_and_wait(); // wait for off-diagonal block
111 if (ν2p(c + 1) + 1 == lp() || p == 1)
112 solve_pcg(λ.batch(0), work_pcg.batch(0));
113 }
114 }
115}

152// Algorithm 5 “Solution of a symmetric block-tridiagonal system using cyclic reduction (CR)”
153// §3.2 Cyclic reduction of block-tridiagonal linear systems
154//
155// The reverse solve routines below closely follow the structure of the corresponding factorization
156// and forward solve routines, but in reverse order. An iterative approach is used instead of
157// recursion. Note that the evaluation of λ(0) is performed during the forward solve step.
158// Depending on the problem size, either a parallel or serial version of the CR solve is used.

165 ctx.arrive_and_wait(); // wait for λ(c-1)

184template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
185void TricyqleSolver<VL, T, DefaultOrder, Ctx>::solve_reverse_parallel(Context &ctx, mut_view<> λ,
186 mut_view<> work,
187 index_t stride) const {
188 const index_t c = ctx.index;
189 for (index_t l = lp(); l-- > 0;) {
190 const auto c_ = cr_thread_assignment(l, c);
191 const index_t i_u = add_wrap_ceil_p(c_, 1), i_y = sub_wrap_ceil_p(c_, (1 << l) - 1);
192 if (l < lp() - 1) { // λ(0) was already computed during forward solve
193 auto wait_uy = ctx.arrive(); // wait for Uᵀλ, Yᵀλ
194 if (ν2p(i_y) == l + 1) {
195 ctx.wait(std::move(wait_uy));
196 solve_λ_backward(i_y, λ, work, stride);
197 } else if (ν2p(i_u) == l) {
198 prefetch_U(l, i_u);
199 ctx.wait(std::move(wait_uy));
200 } else {
201 if (ν2p(i_y) == l)
202 prefetch_Y(l, i_y);
203 ctx.wait(std::move(wait_uy));
204 }
205 }
206 auto wait_λ = ctx.arrive(); // wait for λ
207 if (ν2p(i_u) == l) {
208 ctx.wait(std::move(wait_λ));
209 solve_u_backward(l, i_u, λ, work, stride);
210 } else if (ν2p(i_y) == l) {
211 ctx.wait(std::move(wait_λ));
212 solve_y_backward(l, i_y, λ, stride);
213 } else {
214 if (l > 0) {
215 const auto l_next = l - 1, c_next = cr_thread_assignment(l_next, c);
216 const index_t i_u_next = add_wrap_ceil_p(c_next, 1),
217 i_y_next = sub_wrap_ceil_p(c_next, (1 << l_next) - 1);
218 if (ν2p(i_y_next) == l_next + 1) {
219 prefetch_U(l_next, i_u_next);
220 prefetch_L(i_y_next);
221 }
222 }
223 ctx.wait(std::move(wait_λ));
224 }
225 }
226 ctx.arrive_and_wait(); // wait for Uᵀλ, Yᵀλ
227 if (ν2p(c) == 0 && p != 1)
228 solve_λ_backward(c, λ, work, stride);
229}

233template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
234void TricyqleSolver<VL, T, DefaultOrder, Ctx>::solve_reverse_serial(mut_view<> λ, mut_view<> work,
235 index_t stride) const {
236 for (index_t l = lp(); l-- > 0;) {
237 for (index_t c = 0; c < p; ++c) {
238 const index_t c_ = cr_thread_assignment(l, c);
239 const index_t i_y = sub_wrap_ceil_p(c_, (1 << l) - 1);
240 if (l < lp() - 1) { // λ(0) was already computed during forward solve
241 if (ν2p(i_y) == l + 1)
242 solve_λ_backward(i_y, λ, work, stride);
243 }
244 }
245 for (index_t c = 0; c < p; ++c) {
246 const index_t c_ = cr_thread_assignment(l, c);
247 const index_t i_u = add_wrap_ceil_p(c_, 1), i_y = sub_wrap_ceil_p(c_, (1 << l) - 1);
248 if (ν2p(i_u) == l)
249 solve_u_backward(l, i_u, λ, work, stride);
250 else if (ν2p(i_y) == l)
251 solve_y_backward(l, i_y, λ, stride);
252 }
253 }
254 for (index_t c = 0; c < p; ++c)
255 if (ν2p(c) == 0 && p != 1)
256 solve_λ_backward(c, λ, work, stride);
257}

290/// Adjust thread assignment for non-power-of-two p:
291/// The diagonal blocks M(⌊p/2⌋2) are usually mapped to increasing thread indices c as the CR level
292/// l increases, as can be seen in the functions above, where iY = c + 1 - 2^l, and from the way the
293/// path of M nodes curves to the right in the thread assignment diagram in the paper.
294/// However, these large thread indices are not actually present if p is not a power of two, so
295/// we need to remap them, undoing the offset 1 - 2^l.
296/// We always assign the last M evaluation to the even thread ⌊p/2⌋2, since this thread is present
297/// even if p is odd. The odd thread ⌊p/2⌋2+1 is assigned an inactive index, since it never has any
298/// work during CR, as there is no coupling between the last and first stages (at least not in the
299/// scalar case).
300template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
301index_t TricyqleSolver<VL, T, DefaultOrder, Ctx>::cr_thread_assignment(index_t l, index_t c) const {
302 // Index of the last diagonal block M or L that may need to be handled in this level
303 const auto iL = c & ~index_t{(1 << l) - 1};
304 // Only remap the last two threads: c == p - 1 for odd p; c == p - 2 or c == p - 1 for even p
305 const bool last_threads = (c >> 1) + 1 == (p + 1) >> 1;
306 // If this block iL would be assigned to a thread >= p, remap it to the last even thread < p
307 const bool remap = iL + (1 << l) - 1 >= p;
308 if (!is_pow_2(p) && last_threads && remap)
309 c = c & 1 ? iL // last odd thread gets the inactive index
310 : add_wrap_ceil_p(iL, (1 << l) - 1); // last even thread gets remapped
311 return c;
312}
