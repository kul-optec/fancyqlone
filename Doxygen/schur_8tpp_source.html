31void CyqloneSolver<VL, T, DefaultOrder, Ctx>::compute_schur(Context &ctx, real_t ρ_inv,
32 mut_view<> ux, mut_view<> λ) {
33 const index_t c = riccati_thread_assignment(ctx);
34 const auto c_next = add_wrap_p(c, 1);
35 // 7| j₁ = n(c-1)+1, jₙ = nc
36 const auto dn = c * n, dn_next = c_next * n, d1_next = dn_next + n - 1;
37 // 8| i˃ = c, i˂ = c-1
38 const index_t i_fwd = c, i_bwd = sub_wrap_ceil_p(c, 1);
39 auto M = tril(tricyqle.cr_L.batch(c));
40 // 13| W = [ LB(jₙ) ... LB(j₁) LA(j₁) ] -- The order here is [ LA(j₁) LB(jₙ) ... LB(j₁) ]
41 auto W = riccati_LAB.batch(c).right_cols(nx + nu * n);
42 if constexpr (Factor) {
43 auto LH = riccati_LH.batch(c);
44 auto LQ = tril(LH.bottom_right(nx, nx));
45 // 9| T(c) = LQ(j₁)⁻ᵀ
46 BATMAT_ASSERT(nu >= 1); // T = LQ⁻ᵀ is upper triangular, stored one row up from LQ itself
47 auto Tc = triu(LH.right_cols(nx).middle_rows(nu - 1, nx));
48
49 {
50#if GUANAQO_WITH_TRACING
51
52 GUANAQO_TRACE("Invert Q", c);
53 CYQ_TRACE_WRITE(T, c, 0);
54#endif
55 trtri(LQ, Tc.transposed());
56 }
57 auto T_ready = ctx.arrive();
58 auto LA1 = riccati_LAB.batch(c).middle_cols(nx * (n - 1), nx); // LA(j₁)
59 // 10| if ν2(i˂) > ν2(i˃) K˂(i˃) = -T(c) LA(j₁)ᵀ else K˃(i˂) = -LA(j₁) T(c)ᵀ
60 if (ν2p(i_bwd) > ν2p(i_fwd)) {
61 GUANAQO_TRACE("Compute first U", i_fwd);
62 CYQ_TRACE_WRITE(Kb, i_fwd, 0);
63 trmm_neg(Tc, LA1.transposed(), tricyqle.cr_U.batch(i_fwd));
64 } else {
65 const index_t i_bwd_wrap = sub_wrap_p(c, 1);
66#if GUANAQO_WITH_TRACING
67
68 GUANAQO_TRACE("Compute first Y", i_bwd_wrap);
69 CYQ_TRACE_WRITE(Kf, i_bwd_wrap, 0);
70#endif
71 if (i_fwd > 0 || (v == 1 && !params.eliminate_u0_first))
72 trmm_neg(LA1, Tc.transposed(), tricyqle.cr_Y.batch(i_bwd_wrap));
73 else if constexpr (v > 1)
74 trmm_neg(LA1, Tc.transposed(), tricyqle.cr_Y.batch(i_bwd_wrap), //
75 with_rotate_C<-1>, with_rotate_D<-1>, with_mask_D<-1>);
76 }
77 // 11| -- sync --
78 // Wait for the inversion in the next interval
79 ctx.wait(std::move(T_ready));
80 // Each column of the cyclic part with coupling equations is updated by two threads:
81 // one for the forward, and one for the backward coupling. Update the diagonal blocks
82 // of the coupling equations, first forward in time ...
83 auto R̂ŜQ̂_next = riccati_LH.batch(c_next);
84 // 12| M(c)˂ = T(c+1) T(c+1)ᵀ
85 auto Tc_next = triu(R̂ŜQ̂_next.right_cols(nx).middle_rows(nu - 1, nx));
86 {
87#if GUANAQO_WITH_TRACING
88 CYQ_TRACE_READ(T, c_next, 0);
89 GUANAQO_TRACE("Compute TTᵀ", c_next);
90#endif
91 if (c_next > 0 || v == 1)
92 trmm(Tc_next, Tc_next.transposed(), M);
93 else
94 trmm(Tc_next, Tc_next.transposed(), M, with_rotate_C<-1>, with_rotate_D<-1>);
95 }
96 // And finally backward in time, optionally fused with the factorization.
97 if (p == 1) { // no multi-threading
98#if GUANAQO_WITH_TRACING
99
100 GUANAQO_TRACE("Factor M last", c);
101 CYQ_TRACE_WRITE(L, c, 0);
102#endif
103 auto L0 = tril(tricyqle.pcr_L.batch(0));
104 // 13| M(c)˃ = WWᵀ
105 // 14| M(c) = M(c)˂ + M(c)˃
106 syrk_add(W, M);
107 if (ρ_inv > 0)
108 M.value.add_to_diagonal(ρ_inv);
109 // 16| L(c) = chol(M(c))
110 potrf(M, L0); // Final block is stored separately (for PCR/PCG later)
111 } else if (ν2p(i_fwd) == 0) {
112#if GUANAQO_WITH_TRACING
113
114 GUANAQO_TRACE("Factor M", c);
115 CYQ_TRACE_WRITE(L, c, 0);
116 CYQ_TRACE_WRITE(L, c, 1);
117#endif
118 // 13| M(c)˃ = WWᵀ
119 // 14| M(c) = M(c)˂ + M(c)˃
120 // 16| L(c) = chol(M(c))
121 syrk_add_potrf(W, M, M, ρ_inv);
122 } else {
123#if GUANAQO_WITH_TRACING
124
125 GUANAQO_TRACE("Compute WWᵀ", c);
126 CYQ_TRACE_WRITE(M, c, 0);
127#endif
128
129 // 13| M(c)˃ = WWᵀ
130 // 14| M(c) = M(c)˂ + M(c)˃
131 syrk_add(W, M);
132 if (ρ_inv > 0)
133 M.value.add_to_diagonal(ρ_inv);
134 }
135 }
136 if constexpr (Solve) {
137 if (!Factor)
138 ctx.arrive_and_wait(); // Wait for x_next
139 {
140 GUANAQO_TRACE("Update λ", dn);
141 auto x_next = ux.batch(d1_next).bottom_rows(nx);
142 if (c_next > 0 || v == 1)
143 sub(λ.batch(dn), x_next);
144 else
145 sub(λ.batch(dn), x_next, with_rotate<1>);
146 }
147 {
148 // TODO: λ(dn) here has a different thread assignment than in TricyqleSolver
149 GUANAQO_TRACE("Solve λ", dn);
150 if (ν2p(i_fwd) == 0 && p != 1)
151 trsm(M, λ.batch(dn));
152 }
153 }
154}
