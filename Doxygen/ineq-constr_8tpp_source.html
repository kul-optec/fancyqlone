10template <index_t VL, StorageOrder DefaultOrder, class LinearSolver>
11index_t CyQPALMBackend<VL, DefaultOrder, LinearSolver>::update_penalty_y(
12 Context &ctx, ineq_constr_vec_t &Σ, const ineq_constr_vec_t &e, const ineq_constr_vec_t &e_old,
13 const PenaltySettings &settings) {
14 GUANAQO_TRACE("update_penalty_y", 0);
15 auto t = get_timed(&Timings::update_penalty_y);
16 const real_t min_denom = 1e-6;
17 const real_t norm_inf_e = fmax(min_denom, norm_inf(ctx, e));
18 index_t num_changed = 0;
19 const auto update_simd = [&](auto Σji, auto eji, auto eji_old) {
20 using batmat::datapar::select;
21 using batmat::datapar::reduce_count;
22 auto insufficient_progress = abs(eji) >= settings.θ * abs(eji_old);
23 auto update_factor =
24 select(insufficient_progress, settings.Δy * abs(eji) / norm_inf_e, decltype(eji){1});
25 update_factor *= settings.Δy_always;
26 auto Σ_new = Σji * update_factor;
27 Σ_new =
28 fmax(Σji * settings.Δy_always, fmin(Σ_new, decltype(Σ_new){settings.max_penalty_y}));
29 num_changed += reduce_count(Σ_new != Σji);
30 return Σ_new;
31 };
32 const auto update_batch = [&](auto Σj, auto ej, auto ej_old) {
33 linalg::transform_elementwise(update_simd, Σj, Σj, ej, ej_old);
34 };
35 ocp.foreach_batch(ctx, update_batch, Σ, e, e_old);
36 return ctx.reduce(num_changed);
37}
