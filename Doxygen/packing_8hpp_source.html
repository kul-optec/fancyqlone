26/// Transposes the @p R × @p C matrix at @p pa with leading dimension @p lda, writing the result to
27/// @p pb with leading dimension @p ldb, writing only the @p d first columns of the result.
28template <index_t R, index_t C, class T, int RotIn, int RotOut, bool Add>
29[[gnu::always_inline]]
30inline void transpose_pack(const T *pa, index_t lda, T *pb, index_t ldb) {
31 using batmat::ops::rotl;
32 using simd = batmat::datapar::deduced_simd<T, R>;
33#ifdef __AVX2__
34 if constexpr (std::is_same_v<T, double> && R == 4 && C == 4) {
35 static constexpr auto write_out = [](simd v, T *p) {
36 v = rotl<RotOut>(v);
37 if constexpr (Add)
38 v += batmat::datapar::unaligned_load<simd>(p);
39 batmat::datapar::unaligned_store(v, p);
40 };
41 simd cols[4], shuf[4];
42 BATMAT_FULLY_UNROLLED_FOR (int i = 0; i < 4; ++i)
43 cols[i] = rotl<RotIn>(batmat::datapar::unaligned_load<simd>(pa + i * lda));
44 shuf[0] = simd{_mm256_shuffle_pd((__m256d)cols[0], (__m256d)cols[1], 0b0000)};
45 shuf[1] = simd{_mm256_shuffle_pd((__m256d)cols[0], (__m256d)cols[1], 0b1111)};
46 shuf[2] = simd{_mm256_shuffle_pd((__m256d)cols[2], (__m256d)cols[3], 0b0000)};
47 shuf[3] = simd{_mm256_shuffle_pd((__m256d)cols[2], (__m256d)cols[3], 0b1111)};
48 cols[0] = simd{_mm256_permute2f128_pd((__m256d)shuf[0], (__m256d)shuf[2], 0b00100000)};
49 write_out(cols[0], pb + 0 * ldb);
50 cols[1] = simd{_mm256_permute2f128_pd((__m256d)shuf[1], (__m256d)shuf[3], 0b00100000)};
51 write_out(cols[1], pb + 1 * ldb);
52 cols[2] = simd{_mm256_permute2f128_pd((__m256d)shuf[0], (__m256d)shuf[2], 0b00110001)};
53 write_out(cols[2], pb + 2 * ldb);
54 cols[3] = simd{_mm256_permute2f128_pd((__m256d)shuf[1], (__m256d)shuf[3], 0b00110001)};
55 write_out(cols[3], pb + 3 * ldb);
56 return;
57 }
58#endif
59 simd r[C];
60 BATMAT_FULLY_UNROLLED_FOR (int i = 0; i < C; ++i)
61 r[i] = rotl<RotIn>(batmat::datapar::unaligned_load<simd>(pa + i * lda));
62 BATMAT_FULLY_UNROLLED_FOR (int i = 0; i < R; ++i)
63 BATMAT_FULLY_UNROLLED_FOR (int j = 0; j < C; ++j)
64 if constexpr (Add)
65 pb[i * ldb + j] += r[(j + RotOut + C) % C][i];
66 else
67 pb[i * ldb + j] = r[(j + RotOut + C) % C][i];
68}

75template <class TA, class Abi, int Rot, bool Add, MatrixStructure Struc = MatrixStructure::General,
76 StorageOrder OA, class TB, class DB, class LB, StorageOrder OB>
77 requires(OB == StorageOrder::ColMajor)
78inline void unpack_full(view<TA, Abi, OA> A, scalar_view<TB, DB, LB, OB> B) {
79 static_assert(std::is_const_v<TA> ^ std::is_const_v<TB>);
80 static_assert(typename decltype(B)::batch_size_type() == 1);
81 static constexpr bool Pack = std::is_const_v<TB>;
82 BATMAT_ASSERT(B.rows() == A.rows());
83 BATMAT_ASSERT(B.cols() == A.cols());
84 BATMAT_ASSERT(static_cast<index_t>(B.depth()) == static_cast<index_t>(A.depth()));
85 GUANAQO_TRACE_LINALG(GUANAQO_TRACE_STATIC_STR(Pack ? "pack" : "unpack"),
86 A.rows() * A.cols() * A.depth());
87 using enum MatrixStructure;
88 using T = std::remove_const_t<TA>;
89 static constexpr index_t v = typename decltype(A)::batch_size_type();
90 static constexpr auto lut = batmat::make_1d_lut<v>([]<index_t R>(index_constant<R>) {
91 if constexpr (Pack)
92 return transpose_pack<R + 1, v, T, 0, Rot, Add>;
93 else
94 return transpose_pack<v, R + 1, T, Rot, 0, Add>;
95 });
96 const auto cstrA = A.col_stride() * v;
97 const auto rstrA = A.row_stride() * v;
98 const auto cstrB = B.col_stride();
99 const auto rstrB = B.row_stride();
100 const auto bstrB = B.layer_stride();
101 static_assert(rstrB == 1);
102 TA *pA = A.data();
103 TB *pB = B.data();
104 auto inner_count = Struc == LowerTriangular ? std::max(A.rows(), A.cols())
105 : Struc == UpperTriangular ? 1 + std::max(A.rows(), A.cols()) - A.cols()
106 : A.rows();
107 using std::clamp;
108 for (index_t c = 0; c < A.cols(); ++c) {
109 TA *pA_ = pA;
110 TB *pB_ = pB;
111 batmat::foreach_chunked(
112 0, clamp(inner_count, index_t{0}, A.rows()), v,
113 [&](index_t) {
114 if constexpr (Pack)
115 transpose_pack<v, v, T, 0, Rot, Add>(pB_, bstrB, pA_, rstrA);
116 else
117 transpose_pack<v, v, T, Rot, 0, Add>(pA_, rstrA, pB_, bstrB);
118 pA_ += v * rstrA;
119 pB_ += v * rstrB;
120 },
121 [&](index_t, index_t nr) {
122 if constexpr (Pack)
123 lut[nr - 1](pB_, bstrB, pA_, rstrA);
124 else
125 lut[nr - 1](pA_, rstrA, pB_, bstrB);
126 });
127 pA += cstrA;
128 pB += cstrB;
129 if constexpr (Struc == LowerTriangular) {
130 --inner_count;
131 if (inner_count < A.rows()) {
132 pA += rstrA;
133 pB += rstrB;
134 }
135 } else if (Struc == UpperTriangular) {
136 ++inner_count;
137 }
138 }
139}

150/// @todo Test this function. Then benchmark it against unpack_full: if this variant is not much
151/// slower, we should consider using it for all unpacking so we don't need two variants.
152template <class T, class Abi, StorageOrder OA, class DB, class LB, StorageOrder OB>
153 requires(OA == OB)
154inline void unpack_partial(view<const T, Abi, OA> A, scalar_view<T, DB, LB, OB> B) {
155 if (B.depth() >= A.depth())
156 return unpack_full(A, B.first_layers(A.depth()));
157 static_assert(OA == StorageOrder::ColMajor); // TODO: row major
158 GUANAQO_TRACE_LINALG("unpack", A.rows() * A.cols() * A.depth());
159 static constexpr index_t v = A.batch_size();
160 static constexpr auto lut = batmat::make_1d_lut<v>(
161 []<index_t R>(index_constant<R>) { return batmat::ops::transpose_dyn<v, R + 1, T>; });
162 for (index_t c = 0; c < A.cols(); ++c)
163 batmat::foreach_chunked(
164 0, A.rows(), v,
165 [&](index_t r) {
166 batmat::ops::transpose_dyn<v, v>(&A(0, r, c), v, &B(0, r, c), B.layer_stride(),
167 B.depth());
168 },
169 [&](index_t r, index_t nr) {
170 lut[nr - 1](A.block(r, c, nr, 1).data, v, B.block(r, c, nr, 1).data,
171 B.layer_stride(), B.depth());
172 });
173}
