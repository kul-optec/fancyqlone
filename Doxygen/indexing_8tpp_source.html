143template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
144[[nodiscard]] StageIndexInv
145CyqloneSolver<VL, T, DefaultOrder, Ctx>::inverse_stage_index(index_t stage) const {
146 const index_t N = ceil_N();
147
148 BATMAT_ASSUME(stage >= 0);
149 BATMAT_ASSUME(stage < N);
150 BATMAT_ASSUME(n > 0);
151 BATMAT_ASSUME(N % n == 0);
152
153 const index_t K = N / n; // should be p * VL
154
155 const index_t q = stage / n;
156 const index_t r = stage - q * n;
157
158 index_t k;
159 index_t i;
160
161 if (params.eliminate_u0_first) {
162 // stage = n*k - i mod N
163 if (r == 0) {
164 k = q;
165 i = 0;
166 } else {
167 k = q + 1;
168 if (k == K)
169 k = 0;
170
171 i = n - r;
172 }
173 } else {
174 // stage = n*k + n - 1 - i mod N
175 k = q;
176 i = n - 1 - r;
177 }
178
179 const index_t c = k % p;
180 const index_t l = k / p;
181
182 BATMAT_ASSUME(c >= 0);
183 BATMAT_ASSUME(c < p);
184 BATMAT_ASSUME(l >= 0);
185 BATMAT_ASSUME(l < VL);
186 BATMAT_ASSUME(i >= 0);
187 BATMAT_ASSUME(i < n);
188 BATMAT_ASSERT(stage == stage_index(c, i, l));
189 return {
190 .c = c,
191 .i = i,
192 .l = l,
193 };
194}
