41 index_t block_size = 5; // Size of each block of the block tridiagonal system
42 bool circular = false; // Is K(num_blocks-1) nonzero?
43 index_t rank = 2; // Rank for the factorization updates
44 // Parse arguments
45 if (argc > 1)
46 p = std::stoi(argv[1]);
47 if (argc > 2)
48 block_size = std::stoi(argv[2]);
49 if (argc > 3)
50 circular = std::stoi(argv[3]) != 0;
51 if (argc > 4)
52 rank = std::stoi(argv[4]);
53 // The number of blocks in the block tridiagonal system always equals p * v. If your system is
54 // larger, first reduce it to this size (in parallel). If it is smaller, manually add padding,
55 // or reduce p and/or v.
56 BATMAT_ASSERT(fancyqlone::is_pow_2(p));
57 const index_t num_blocks = p * v;

62 // Create a solver for the block tridiagonal system
63 TricyqleSolver solver{
64 .block_size = block_size, .max_rank = p * rank, .circular = circular, .p = p};
65 solver.params.solve_method = fancyqlone::SolveMethod::PCR; // Use the PCR solver instead of PCG.
66
67 // Create a compact copy of the right-hand side and allocate storage for the solution
68 TricyqleSolver::matrix b_solve{{.depth = num_blocks, .rows = block_size, .cols = 1}};
69 matrices x{{.depth = num_blocks, .rows = block_size, .cols = 1}};
70
71 // Call the solver in parallel, passing a lambda that is executed by each thread.
72 auto pctx = solver.create_parallel_context();
73 pctx->run([&](TricyqleSolver::Context &ctx) {
74 // Copy the data to the solver's internal data structures
75 using fancyqlone::linalg::pack;
76 auto pack_M = [&](index_t i, auto Ms) { return pack(M.middle_batches(i, v, p), Ms); };
77 auto pack_K = [&](index_t i, auto Ks) { return pack(K.middle_batches(i, v, p), Ks); };
78 auto pack_b = [&](index_t i, auto bs) { return pack(b.middle_batches(i, v, p), bs); };

81 solver.init_rhs(ctx, b_solve, pack_b);
82 // Perform the factorization and the forward solve (fused), followed by the backward solve
83 solver.factor_solve(ctx, b_solve);
84 solver.solve_reverse(ctx, b_solve);
85 // Copy the solution back to the original layout
86 using fancyqlone::linalg::unpack;
87 auto unpack_x = [&](index_t i, auto xs) { return unpack(xs, x.middle_batches(i, v, p)); };
88 solver.get_solution(ctx, b_solve, unpack_x);
89 }); // blocks until all threads have joined
90
91 // Perform a factorization update and solve again
92 auto [B, F, S] = init_random_update(block_size, num_blocks, rank, circular);
93 matrices x2{{.depth = num_blocks, .rows = block_size, .cols = 1}};
94 pctx->run([&](TricyqleSolver::Context &ctx) {
95 using fancyqlone::linalg::pack;
96 auto pack_b = [&](index_t i, auto bs) { return pack(b.middle_batches(i, v, p), bs); };
97 auto pack_upd = [&](auto iS, auto Ss, auto &sync_S, auto iB, auto Bs, auto iF, auto Fs) {
98 pack(S.middle_batches(iS, v, p), Ss); // diagonal update matrix
99 sync_S(); // indicate that our part of S is ready (needs to be exchanged across threads)
100 pack(B.middle_batches(iB, v, p), Bs); // diagonal blocks of the update matrix
101 pack(F.middle_batches(iF, v, p), Fs); // subdiagonal blocks of the update matrix

104 solver.init_rhs(ctx, b_solve, pack_b);
105 solver.update_solve_cr(ctx, b_solve);

162TridiagSystem init_random_system(index_t block_size, index_t num_blocks, bool circular) {
163 using batmat::matrix::uninitialized;
164 std::mt19937 rng(12345);
165 std::normal_distribution<real_t> dist(0.0, 1.0);
166 // Generate a random lower block bidiagonal matrix with diagonal blocks A(i) and subdiagonal
167 // blocks B(i).
168 matrices A{{.depth = num_blocks, .rows = block_size, .cols = block_size}, uninitialized};
169 matrices B{{.depth = num_blocks, .rows = block_size, .cols = block_size}, uninitialized};
170 std::ranges::generate(A, [&] { return dist(rng); });
171 std::ranges::generate(B, [&] { return dist(rng); });
172 // Allocate storage for a block tridiagonal system
173 matrices M{{.depth = num_blocks, .rows = block_size, .cols = block_size}, uninitialized};
174 matrices K{{.depth = num_blocks, .rows = block_size, .cols = block_size}, uninitialized};
175 matrices b{{.depth = num_blocks, .rows = block_size, .cols = 1}, uninitialized};
176 std::ranges::generate(b, [&] { return dist(rng); });
177 if (!circular)
178 B(num_blocks - 1).set_constant(0);
179 // Diagonal blocks M(i) = A(i) A(i)ᵀ + B(i-1) B(i-1)ᵀ and subdiagonal blocks K(i) = B(i) A(i)ᵀ
180 for (index_t i = 0; i < num_blocks; ++i) {
181 index_t i_prev = (i - 1 + num_blocks) % num_blocks;
182 guanaqo::blas::xsyrk_LN<real_t>(1, A(i), 0, M(i)); // M(i) = A(i) A(i)ᵀ
183 guanaqo::blas::xsyrk_LN<real_t>(1, B(i_prev), 1, M(i)); // M(i) += B(i-1) B(i-1)ᵀ
184 guanaqo::blas::xgemm_NT<real_t>(1, B(i), A(i), 0, K(i)); // K(i) = B(i) A(i)ᵀ
185 M(i).add_to_diagonal(1e-4); // Ensure positive definiteness
186 }
187 return TridiagSystem{.M = std::move(M), .K = std::move(K), .b = std::move(b)};
188}

190BidiagUpdate init_random_update(index_t block_size, index_t num_blocks, index_t rank,
191 bool circular) {
192 using batmat::matrix::uninitialized;
193 std::mt19937 rng(54321);
194 std::normal_distribution<real_t> dist(0.0, 1.0);
195 // Generate a random bidiagonal matrix with diagonal blocks B(i) and subdiagonal blocks F(i), as
196 // well as random diagonal matrices Σ.
197 matrices B{{.depth = num_blocks, .rows = block_size, .cols = rank}, uninitialized};
198 matrices F{{.depth = num_blocks, .rows = block_size, .cols = rank}, uninitialized};
199 matrices S{{.depth = num_blocks, .rows = rank, .cols = 1}, uninitialized};
200 std::ranges::generate(B, [&] { return dist(rng); });
201 std::ranges::generate(F, [&] { return dist(rng); });
202 std::ranges::generate(S, [&] { return std::exp2(dist(rng)); });
203 if (!circular)
204 F(num_blocks - 1).set_constant(0); // No circular coupling
205 return BidiagUpdate{.B = std::move(B), .F = std::move(F), .S = std::move(S)};
206}
