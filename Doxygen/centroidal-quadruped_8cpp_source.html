32Vector3 reference_position(const CentroidalQuadrupedParams &p, real_t time) {
33 const auto [vx, vy, yaw_rate] = p.v_ref;
34 Vector3 position{0, 0, p.base_height};
35 if (yaw_rate != 0) {
36 const auto c = std::cos(yaw_rate * time);
37 const auto s = std::sin(yaw_rate * time);
38 position(0) = (vx * s + vy * (c - 1)) / yaw_rate;
39 position(1) = (vy * s - vx * (c - 1)) / yaw_rate;
40 } else {
41 position(0) = vx * time;
42 position(1) = vy * time;
43 }
44 return position;
45}

47void set_reference_state(const CentroidalQuadrupedParams &p, real_t time, real_t *state) {
48 const auto [vx, vy, yaw_rate] = p.v_ref;
49 const auto position = reference_position(p, time);
50 state[0] = position(0);
51 state[1] = position(1);
52 state[2] = position(2);
53 state[5] = yaw_rate * time;
54 state[11] = yaw_rate;
55 if (yaw_rate != 0) {
56 const auto c = std::cos(yaw_rate * time);
57 const auto s = std::sin(yaw_rate * time);
58 state[6] = vx * c + vy * s;
59 state[7] = vy * c - vx * s;
60 } else {
61 state[6] = vx;
62 state[7] = vy;
63 }
64}

66Matrix3 skew(const Vector3 &v) {
67 Matrix3 result;
68 result << 0, -v(2), v(1), v(2), 0, -v(0), -v(1), v(0), 0;

72std::vector<std::array<bool, 4>> expand_gait(const CentroidalQuadrupedParams &p) {
73 std::vector<std::array<bool, 4>> contacts;
74 contacts.reserve(static_cast<size_t>(p.N_horiz));
75 index_t cycle_duration = 0;
76 for (const auto &phase : p.gait) {
77 if (phase.duration <= 0)
78 throw std::invalid_argument("Gait phase durations must be positive");
79 if (phase.duration > std::numeric_limits<index_t>::max() - cycle_duration)
80 throw std::invalid_argument("Gait cycle duration is too large");
81 cycle_duration += phase.duration;
82 }
83 auto offset = p.initial_stage % cycle_duration;
84 while (static_cast<index_t>(contacts.size()) < p.N_horiz) {
85 for (const auto &phase : p.gait) {
86 if (offset >= phase.duration) {
87 offset -= phase.duration;
88 continue;
89 }
90 for (index_t i = offset;
91 i < phase.duration && static_cast<index_t>(contacts.size()) < p.N_horiz; ++i)
92 contacts.push_back(phase.contacts);
93 offset = 0;
94 }
95 }
96 return contacts;
97}

99void validate(const CentroidalQuadrupedParams &p) {
100 if (p.gait.empty())
101 throw std::invalid_argument("The gait must contain at least one phase");
102 if (p.N_horiz <= 0)
103 throw std::invalid_argument("The prediction horizon must be positive");
104 if (p.initial_stage < 0)
105 throw std::invalid_argument("The initial prediction stage must be nonnegative");
106 if (p.dt <= 0)
107 throw std::invalid_argument("The time step must be positive");
108 if (p.mass <= 0)
109 throw std::invalid_argument("The robot mass must be positive");
110 if (p.mu < 0)
111 throw std::invalid_argument("The friction coefficient must be nonnegative");
112 if (p.f_max < 0)
113 throw std::invalid_argument("The maximum contact force must be nonnegative");
114}

118CentroidalQuadrupedProblem centroidal_quadruped(CentroidalQuadrupedParams p) {
119 using guanaqo::as_view;
120 validate(p);
121 const auto contacts = expand_gait(p);
122 const auto N = static_cast<index_t>(contacts.size());
123 const auto inf = std::numeric_limits<real_t>::infinity();
124
125 LinearOCPStorage ocp{.dim{
126 .N_horiz = N,
127 .nx = nx,
128 .nu = nu,
129 .ny = ny,
130 .ny_N = 0,
131 }};
132 ocp.b().set_constant(0);
133 ocp.b(0)(2, 0) = p.base_height;
134 ocp.b_min().set_constant(-inf);
135 ocp.b_max().set_constant(+inf);
136
137 std::vector<real_t> reference(static_cast<size_t>(N * (nx + nu) + nx));
138 for (index_t i = 0; i <= N; ++i) {
139 auto offset = static_cast<size_t>(i * (nx + nu));
140 const auto time = (static_cast<real_t>(p.initial_stage) + static_cast<real_t>(i)) * p.dt;
141 set_reference_state(p, time, reference.data() + offset);
142 }
143
144 Matrix3 inertia;
145 inertia << p.inertia[0], p.inertia[1], p.inertia[2], p.inertia[3], p.inertia[4], p.inertia[5],
146 p.inertia[6], p.inertia[7], p.inertia[8];
147 const std::array<Vector3, 4> footholds{
148 Vector3{+p.foot_x, +p.foot_y, 0},
149 Vector3{+p.foot_x, -p.foot_y, 0},
150 Vector3{-p.foot_x, +p.foot_y, 0},
151 Vector3{-p.foot_x, -p.foot_y, 0},
152 };
153
154 for (index_t i = 0; i < N; ++i) {
155 auto Ai = ocp.A(i), Bi = ocp.B(i), Di = ocp.D(i);
156 auto Qi = ocp.Q(i), Ri = ocp.R(i);
157 auto lbi = ocp.b_min(i), ubi = ocp.b_max(i);
158
159 Ai.set_constant(0);
160 Ai.set_diagonal(1);
161 for (index_t j = 0; j < 6; ++j)
162 Ai(j, j + 6) = p.dt;
163
164 for (index_t foot = 0; foot < 4; ++foot)
165 for (index_t axis = 0; axis < 3; ++axis)
166 Bi(6 + axis, 3 * foot + axis) = p.dt / p.mass;
167
168 const auto ref_offset = static_cast<size_t>((i + 1) * (nx + nu));
169 const auto yaw = reference[ref_offset + 5];
170 const auto c = std::cos(yaw);
171 const auto s = std::sin(yaw);
172 Matrix3 rotation;
173 rotation << c, -s, 0, s, c, 0, 0, 0, 1;
174 const Matrix3 inertia_inv = (rotation.transpose() * inertia * rotation).inverse();
175 const Vector3 base_position{reference[ref_offset], reference[ref_offset + 1],
176 reference[ref_offset + 2]};
177 for (index_t foot = 0; foot < 4; ++foot) {
178 const Matrix3 angular = inertia_inv * skew(footholds[foot] - base_position) * p.dt;
179 Bi.block(9, 3 * foot, 3, 3) = as_view(angular);
180 }
181 ocp.b(i + 1)(8, 0) = -p.gravity * p.dt;
182
183 if (i > 0)
184 for (index_t j = 0; j < nx; ++j)
185 Qi(j, j) = p.state_weights[j];
186 for (index_t j = 0; j < nu; ++j)
187 Ri(j, j) = p.force_weights[j];
188
189 for (index_t foot = 0; foot < 4; ++foot) {
190 const auto row = 4 * foot;
191 const auto col = 3 * foot;
192 Di(row + 0, col + 0) = +1;
193 Di(row + 1, col + 0) = -1;
194 Di(row + 2, col + 1) = +1;
195 Di(row + 3, col + 1) = -1;
196 for (index_t j = 0; j < 4; ++j) {
197 Di(row + j, col + 2) = -p.mu;
198 ubi(row + j, 0) = 0;
199 }
200 for (index_t axis = 0; axis < 3; ++axis) {
201 const auto force_row = 16 + col + axis;
202 Di(force_row, col + axis) = 1;
203 if (!contacts[i][foot]) {
204 lbi(force_row, 0) = 0;
205 ubi(force_row, 0) = 0;
206 } else if (axis == 2) {
207 lbi(force_row, 0) = 0;
208 ubi(force_row, 0) = p.f_max;
209 }
210 }
211 }
212 }
213
214 auto QN = ocp.Q(N);
215 for (index_t j = 0; j < nx; ++j)
216 QN(j, j) = p.state_weights[j];
217 reference_to_gradient(ocp, reference);
218
219 return {.ocp = std::move(ocp), .reference = std::move(reference)};
220}

222FanCentroidalQuadrupedProblem fan_centroidal_quadruped(CentroidalQuadrupedParams p,
223 index_t num_branches, real_t mu_radius) {
224 if (num_branches <= 0)
225 throw std::invalid_argument("The number of branches must be positive");
226 if (mu_radius < 0 || p.mu - mu_radius < 0)
227 throw std::invalid_argument("The friction uncertainty interval must be nonnegative");
228
229 std::vector<LinearOCPStorage> ocps;
230 std::vector<real_t> friction_coefficients;
231 const auto nominal_mu = p.mu;
232 ocps.reserve(static_cast<size_t>(num_branches));
233 friction_coefficients.reserve(static_cast<size_t>(num_branches));
234 for (index_t branch = 0; branch < num_branches; ++branch) {
235 if (num_branches > 1)
236 p.mu =
237 nominal_mu - mu_radius +
238 2 * mu_radius * static_cast<real_t>(branch) / static_cast<real_t>(num_branches - 1);
239 friction_coefficients.push_back(p.mu);
240 ocps.push_back(std::move(centroidal_quadruped(p).ocp));
241 }
242 return {
243 .ocp = FanOCPStorage{.ocps = std::move(ocps)},
244 .friction_coefficients = std::move(friction_coefficients),
245 };
246}
