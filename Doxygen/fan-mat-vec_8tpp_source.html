188template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
189void FanCyqloneSolver<VL, T, DefaultOrder, Ctx>::cost_gradient_remove_regularization(
190 Context &ctx, value_type γ, const FanStorage &ux, const FanStorage &ux0,
191 FanStorage &grad_f) const {
192 auto inv_γ{1 / γ};
193 const auto sub_reg_simd = [inv_γ](auto grad_fji, auto xji, auto x0ji) {
194 return grad_fji + inv_γ * (x0ji - xji);
195 };
196 const auto sub_reg = [&]([[maybe_unused]] auto uxj, auto ux0j, auto grad_fj) {
197 linalg::transform_elementwise(sub_reg_simd, grad_fj, grad_fj, uxj, ux0j);
198 };
199 foreach_batch(ctx, sub_reg, ux, ux0, grad_f);
200}
