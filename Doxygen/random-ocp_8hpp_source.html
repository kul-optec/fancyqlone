35 Hi(c, r) = Hi(r, c);
36 auto qi = ocp.q(i), ri = ocp.r(i), bi = ocp.b(i);
37 qi.generate([&] { return uni(rng); });

39 bi.generate([&] { return uni(rng); });

47 for (index_t r = c + 1; r < nx; ++r)
48 Qi(c, r) = Qi(r, c);
49 auto qi = ocp.q(N), bi = ocp.b(N);
50 qi.generate([&] { return uni(rng); });
51 bi.generate([&] { return uni(rng); });

58 x_rand.left_cols(1) = ocp.b(0); // x0

65 for (index_t k = 0; k < nu; ++k)

67 ocp.b_max(i)(j, 0) = ocp.b_min(i)(j, 0);
68 // Optionally add some slack to the constraints

70 ocp.b_min(i)(j, 0) -= abs(uni(rng));
71 if (bernoulli(rng))
72 ocp.b_max(i)(j, 0) += abs(uni(rng));
73 }
74 // Propagate dynamics
75 for (index_t j = 0; j < nx; ++j) {
76 xi_next(j, 0) = 0;
77 for (index_t k = 0; k < nx; ++k)
78 xi_next(j, 0) += ocp.A(i)(j, k) * xi(k, 0);
79 for (index_t k = 0; k < nu; ++k)
