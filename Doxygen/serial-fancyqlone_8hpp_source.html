110 /// Get the total number of primal variables in the OCP.
111 /// @note The actual number of variable stored in Cyqlone's internal data structures may be
112 /// larger.
113 [[nodiscard]] index_t num_variables() const {
114 index_t num_var = 0;
115 for (const SolverStorage<VL, value_type, DefaultOrder, parallel::Context<>> &s :
116 solver_storages) {
117 num_var += nu * (s.cyqlone_solver.N_horiz - 1) +
118 s.cyqlone_solver.nx * s.cyqlone_solver.N_horiz;
119 }
120 return num_var + nu; // for u(0)
121 }
122 /// Get the total number of dynamics constraints in the OCP.
123 /// @note The actual number of constraints stored in Cyqlone's internal data structures may be
124 /// larger.
125 [[nodiscard]] index_t num_dynamics_constraints() const {
126 index_t num_dyn_con = 0;
127 for (const SolverStorage<VL, value_type, DefaultOrder, parallel::Context<>> &s :
128 solver_storages) {
129 num_dyn_con += s.cyqlone_solver.nx * s.cyqlone_solver.N_horiz;
130 }
131 return num_dyn_con;
132 }
133 /// Get the total number of general constraints in the OCP.
134 /// @note The actual number of constraints stored in Cyqlone's internal data structures may be
135 /// larger.
136 [[nodiscard]] index_t num_general_constraints() const {
137 index_t num_gen_con = 0;
138 for (const SolverStorage<VL, value_type, DefaultOrder, parallel::Context<>> &s :
139 solver_storages) {
140 num_gen_con +=
141 s.cyqlone_solver.ny * (s.cyqlone_solver.N_horiz - 1) + s.cyqlone_solver.ny_N;
142 }
143 return num_gen_con + ny_0; // for D(0) u(0)
144 }
145 [[nodiscard]] index_t num_internal_constraints() const {
146 index_t num_gen_con = 0;
147 for (const SolverStorage<VL, value_type, DefaultOrder, parallel::Context<>> &s :
148 solver_storages) {
149 const index_t ny_M =
150 std::max(s.cyqlone_solver.ny, s.cyqlone_solver.ny_0 + s.cyqlone_solver.ny_N);
151 num_gen_con += ny_M * (s.cyqlone_solver.ceil_N());
152 }
153 return num_gen_con + ny_0; // for D(0) u(0)
154 }
