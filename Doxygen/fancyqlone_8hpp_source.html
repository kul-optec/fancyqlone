94 FanStorage_t<matrix, value_type, DefaultOrder> clone() const {
95 FanStorage_t<matrix, value_type, DefaultOrder> copy;
96 copy.branches.resize(branches.size());
97 for (index_t b = 0; b < static_cast<index_t>(branches.size()); ++b) {
98 copy.branches[b] = matrix{{.depth = branches[b].depth(),
99 .rows = branches[b].rows(),
100 .cols = branches[b].cols()},
101 batmat::matrix::uninitialized};
102 batmat::linalg::copy(branches[b], copy.branches[b]);
103 }
104 copy.first =
105 matrix_f{{.rows = first.rows(), .cols = first.cols()}, batmat::matrix::uninitialized};
106 batmat::linalg::copy(first, copy.first);
107 return copy;

139 /// Get the total number of primal variables in the OCP.
140 /// @note The actual number of variable stored in Cyqlone's internal data structures may be
141 /// larger.
142 [[nodiscard]] index_t num_variables() const {
143 index_t num_var = 0;
144 for (const SolverStorage<VL, value_type, DefaultOrder, parallel::Context<>> &s :
145 solver_storages) {
146 num_var += nu * (s.cyqlone_solver.N_horiz - 1) +
147 s.cyqlone_solver.nx * s.cyqlone_solver.N_horiz;
148 }
149 return num_var + nu; // for u(0)
150 }
151 /// Get the total number of dynamics constraints in the OCP.
152 /// @note The actual number of constraints stored in Cyqlone's internal data structures may be
153 /// larger.
154 [[nodiscard]] index_t num_dynamics_constraints() const {
155 index_t num_dyn_con = 0;
156 for (const SolverStorage<VL, value_type, DefaultOrder, parallel::Context<>> &s :
157 solver_storages) {
158 num_dyn_con += s.cyqlone_solver.nx * s.cyqlone_solver.N_horiz;
159 }
160 return num_dyn_con;
161 }
162 /// Get the total number of general constraints in the OCP.
163 /// @note The actual number of constraints stored in Cyqlone's internal data structures may be
164 /// larger.
165 [[nodiscard]] index_t num_general_constraints() const {
166 index_t num_gen_con = 0;
167 for (const SolverStorage<VL, value_type, DefaultOrder, parallel::Context<>> &s :
168 solver_storages) {
169 num_gen_con +=
170 s.cyqlone_solver.ny * (s.cyqlone_solver.N_horiz - 1) + s.cyqlone_solver.ny_N;
171 }
172 return num_gen_con + ny_0; // for D(0) u(0)
173 }
174 [[nodiscard]] index_t num_internal_constraints() const {
175 index_t num_gen_con = 0;
176 for (const SolverStorage<VL, value_type, DefaultOrder, parallel::Context<>> &s :
177 solver_storages) {
178 const index_t ny_M =
179 std::max(s.cyqlone_solver.ny, s.cyqlone_solver.ny_0 + s.cyqlone_solver.ny_N);
180 num_gen_con += ny_M * (s.cyqlone_solver.ceil_N());
181 }
182 return num_gen_con + ny_0; // for D(0) u(0)
183 }
