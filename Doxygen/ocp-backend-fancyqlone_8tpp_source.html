147 void warm_start(const var_vec_t &x, const ineq_constr_vec_t &y, const eq_constr_vec_t &λ) {
148 if constexpr (std::is_same_v<typename traits::warm_start_t, WarmStartingStrategy>) {
149
150 // std::cout << "Warm starting with strategy: " << static_cast<int>(settings.strategy) << std::endl;
151 switch (settings.strategy) {
152 case WarmStartingStrategy::Zeros:
153 this->x0.reset();
154 this->λ0.reset();
155 this->y0.reset();
156 break;
157 case WarmStartingStrategy::Copy:
158 this->x0 = std::move(x);
159 this->λ0 = std::move(λ);
160 this->y0 = std::move(y);
161 break;
162 case WarmStartingStrategy::Shift: {
163 // std::cout << "Warm starting with Shift strategy." << std::endl;
164
165 auto &x0 = this->x0.emplace(var_vec()); // TODO: zero init is redundant
166 auto &λ0 = this->λ0.emplace(eq_constr_vec()); // TODO: zero init is redundant
167 auto &y0 = this->y0.emplace(ineq_constr_vec()); // TODO: zero init is redundant
168 this->ocp.shift_solution(x, x0);
169 this->ocp.shift_dynamics(λ, λ0);
170 this->ocp.shift_constraints(y, y0);
171 } break;
172 case WarmStartingStrategy::ShiftNoInequality: {
173 auto &x0 = this->x0.emplace(var_vec()); // TODO: zero init is redundant
174 auto &λ0 = this->λ0.emplace(eq_constr_vec()); // TODO: zero init is redundant
175 this->y0.emplace(y);
176 this->ocp.shift_solution(x, x0);
177 this->ocp.shift_dynamics(λ, λ0);
178 } break;
179 default: BATMAT_ASSERT(false);
180 }
181 } else if constexpr (std::is_same_v<typename traits::warm_start_t,
182 FanWarmStartingStrategy>) {
183
184 switch (settings.strategy) {
185 case WarmStartingStrategy::Zeros:
186 this->x0.reset();

442 real_t unscaled_aug_lagr_norm(Context &ctx, const var_vec_t &grad_f, const var_vec_t &Mᵀλ,
443 const var_vec_t &Aᵀŷ) const {
444
445 GUANAQO_TRACE("unscaled_aug_lagr_norm", 0);
446 static constexpr auto norms_simd = batmat::linalg::norms<real_t, typename traits::simd>{};
447 using simd_scal = batmat::datapar::simd<real_t, batmat::datapar::scalar_abi<real_t>>;
448 static constexpr auto norms_smid_scal = batmat::linalg::norms<real_t, simd_scal>{};
449 static constexpr auto norms_scal = batmat::linalg::norms<real_t>{};
450
451 auto nrm_simd = norms_simd.zero_simd();
452 auto nrm_simd_scal = norms_smid_scal.zero_simd();
453 const auto grad_norm_simd = [&](auto grad_fji, auto Mᵀλji, auto Aᵀŷji) {
454 if constexpr (std::is_same_v<decltype(grad_fji), simd_scal>) {
455 nrm_simd_scal = norms_smid_scal(nrm_simd_scal, grad_fji + Mᵀλji + Aᵀŷji);
456 } else {
457 nrm_simd = norms_simd(nrm_simd, grad_fji + Mᵀλji + Aᵀŷji);
458 }
459 };
460 const auto grad_norm_batch = [&](auto grad_fj, auto Mᵀλj, auto Aᵀŷj) {
461 linalg::for_each_elementwise(grad_norm_simd, grad_fj, Mᵀλj, Aᵀŷj);
462 // guanaqo::print_python(std::cout << "unscaled_aug_lagr_norm: " << grad_fj(0) << std::endl);
463 };
464 ocp.foreach_batch(ctx, grad_norm_batch, grad_f, Mᵀλ, Aᵀŷ);
465 auto nrm_scal = norms_smid_scal(nrm_simd_scal);
466 nrm_scal = norms_scal(nrm_scal, norms_simd(nrm_simd));
467 // std::cout << "nrm_simd: " << nrm_simd.norm_inf() << ", nrm_scal: " << nrm_scal.norm_inf() << std::endl;
468 // auto nrms_scal = norms_scal.zeros();
469 real_t nrm = ctx.reduce(nrm_scal, norms_scal).norm_inf();
470 // ocp.print_grad(ctx, grad_f);
471 // ocp.print_grad(ctx, Mᵀλ);
472 // std::cout << "unscaled_aug_lagr_norm: " << nrm << std::endl;
473 return nrm;
474 }

499 index_t active_set_change(Context &ctx, real_t, [[maybe_unused]] const ineq_constr_vec_t &Σ,
500 const active_set_t &J, const active_set_t &J_old) {
501 J.check_dimensions_col(J_old, 1);
502 index_t num_different = 0;
503 const auto active_set_change = [&](auto Ji, auto J_oldi) {
504 num_different += std::inner_product(Ji.data(), Ji.data() + Ji.size(), J_oldi.data(), //
505 index_t{0}, std::plus<>{}, std::not_equal_to<>{});
506 };
507 {
508 GUANAQO_TRACE("active_set_change", 0);
509 auto t = get_timed(&Timings::update_active_set_change);
510 ocp.foreach_batch(ctx, active_set_change, J, J_old);
511 }
512 num_different = ctx.reduce(num_different);
513 // If there are no changing constraints, or if we were going to
514 // re-factorize anyway, we don't need to do anything.
515 if (num_different == 0 || reset_factorization)
516 return num_different;
517 bool do_reset_fac =
518 !traits::supports_factorization_updates || num_updates >= settings.max_update_count;
519 do_reset_fac |= static_cast<double>(num_different) >=
520 static_cast<double>(num_ineq_constr()) * settings.changing_constr_factor;
521 bool prev_update_pending = update_pending;
522 if (do_reset_fac) {
523 ctx.run_single_sync([&] {
524 reset_factorization = true;
525 update_pending = false;
526 });
527 return num_different;
528 }
529 ctx.run_single_sync([&] {
530 update_pending = true;
531 ++num_updates;
532 ++stats.num_updates;
533 stats.rank_updates += num_different;
534 });
535 auto t = get_timed(&Timings::update_factorization);
536 if (prev_update_pending) {
537 const auto delta = [&](auto Ji, auto J_oldi, auto ΔΣi) {
538 linalg::axpy(ΔΣi, {1, -1}, Ji, J_oldi);
539 };
540 ocp.foreach_batch(ctx, delta, J, J_old, ΔΣ);
541 } else {
542 const auto delta = [&](auto Ji, auto J_oldi, auto ΔΣi) {
543 linalg::sub(Ji, J_oldi, ΔΣi);
544 };
545 ocp.foreach_batch(ctx, delta, J, J_old, ΔΣ);
546 }
547 return num_different;
548 }
