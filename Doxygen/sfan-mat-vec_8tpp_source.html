133 FanStorage &grad_f) const {
134 auto inv_γ{1 / γ};
135 const auto reg_simd = [inv_γ](auto qji, auto xji, auto x0ji) {

151template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
152void SerialFanCyqloneSolver<VL, T, DefaultOrder, Ctx>::cost_gradient_remove_regularization(
153 Context &ctx, value_type γ, const FanStorage &ux, const FanStorage &ux0,
154 FanStorage &grad_f) const {
155 auto inv_γ{1 / γ};
156 const auto sub_reg_simd = [inv_γ](auto grad_fji, auto xji, auto x0ji) {
157 return grad_fji + inv_γ * (x0ji - xji);
158 };
159 const auto sub_reg = [&]([[maybe_unused]] auto uxj, auto ux0j, auto grad_fj) {
160 linalg::transform_elementwise(sub_reg_simd, grad_fj, grad_fj, uxj, ux0j);
161 };
162 foreach_batch(ctx, sub_reg, ux, ux0, grad_f);
163}
