175template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
176void FanCyqloneSolver<VL, T, DefaultOrder, Ctx>::pack_variables(std::span<const value_type> ux_lin,
177 FanStorage &ux) const {
178 BATMAT_ASSERT(static_cast<index_t>(ux_lin.size()) == num_variables());
179 BATMAT_ASSERT(static_cast<index_t>(ux.branches.size()) == num_branches);
180 index_t offset = 0;
181 for (index_t c = 0; c < num_branches; ++c) {
182 const auto &solver = solver_storages[c].cyqlone_solver;
183 const index_t nux = nu + solver.nx, N_horiz = solver.N_horiz;
184
185 BATMAT_ASSERT(ux.branches[c].depth() == solver.ceil_N());
186 BATMAT_ASSERT(ux.branches[c].rows() == nux);
187 BATMAT_ASSERT(ux.branches[c].cols() == 1);
188 // The compact layout stores u0 only once. Overlap the first nu entries
189 // of each span with the preceding data, then clear the dummy control
190 // that the single-branch packer reads from those entries.
191 solver.pack_variables(ux_lin.subspan(offset, nux * N_horiz), ux.branches[c]);
192 const auto root = solver.inverse_stage_index(0);
193 ux.branches[c].batch(root.c * solver.n + root.i)(root.l).top_rows(nu).set_constant(0);
194 offset += nux * N_horiz - nu;
195 }
196 using crview = guanaqo::MatrixView<const value_type, index_t>;
197 la::copy(crview::as_column(ux_lin.first(nu)), ux.first.batch(0).top_rows(nu));
198}
199template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
200void FanCyqloneSolver<VL, T, DefaultOrder, Ctx>::unpack_variables(
201 const FanStorage &ux, std::span<value_type> ux_lin) const {
202 BATMAT_ASSERT(static_cast<index_t>(ux.branches.size()) == num_branches);
203
204 index_t offset = static_cast<index_t>(ux_lin.size()); //put u0_sol at the end of ux_lin
205
206 // Copy u0_sol into ux_lin
207 using rview = guanaqo::MatrixView<value_type, index_t>;
208
209 // copy the rest of the variables from each branch into ux_lin but leave out the dummy u0
210 for (index_t c = num_branches - 1; c >= 0; --c) {
211 index_t nux = nu + solver_storages[c].cyqlone_solver.nx;
212 index_t ceil_N = solver_storages[c].cyqlone_solver.ceil_N();
213 index_t N_horiz = solver_storages[c].cyqlone_solver.N_horiz;
214
215 BATMAT_ASSERT(ux.branches[c].depth() == ceil_N);
216 BATMAT_ASSERT(ux.branches[c].rows() == nux);
217 BATMAT_ASSERT(ux.branches[c].cols() == 1);
218 solver_storages[c].cyqlone_solver.unpack_variables(
219 ux.branches[c], ux_lin.subspan(offset - nux * N_horiz, nux * N_horiz));
220 offset -= nux * N_horiz - nu;
221 }
222 la::copy(
223 ux.first.batch(0).top_rows(nu),
224 rview::as_column(ux_lin.first(
225 nu))); // copy u0_sol from the first element of ux into the first nu elements of ux_lin
226}
