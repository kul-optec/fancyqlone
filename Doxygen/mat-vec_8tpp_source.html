17void CyqloneSolver<VL, T, DefaultOrder, Ctx>::residual_dynamics_constr(Context &ctx, view<> x,
18 view<> b,
19 mut_view<> Mxb) const {
20 // (Mx + b)(j) = A(j) x(j) + B(j) u(j) - x(j+1) + b(j)
21 auto arrival = ctx.arrive();
22 const index_t c = riccati_thread_assignment(ctx);
23 const index_t dn = c * n; // data batch index
24 const index_t c_next = add_wrap_p(c, 1);
25 const index_t dn_next = c_next * n, d1_next = dn_next + n - 1;
26 for (index_t i = n; i-- > 0;) {
27 auto j = stage_index(c, i, 0);
28 GUANAQO_TRACE("resid_dyn_constr", j);
29 index_t di = dn + i;
30 auto BAj = data_F.batch(di), Bj = BAj.left_cols(nu);
31 auto uxj = x.batch(di), uj = uxj.top_rows(nu);
32 auto bj = b.batch(di);
33 auto Mxbj = Mxb.batch(di);
34 if (v == 1 && j == 0)
35 gemv_add(Bj, uj, bj, Mxbj); // A(j) x(j) + B(j) u(j) + b(j)
36 else
37 gemv_add(BAj, uxj, bj, Mxbj); // A(j) x(j) + B(j) u(j) + b(j)
38 if (i > 0) {
39 index_t di_next = di - 1; // j + 1
40 auto x_next = x.batch(di_next).bottom_rows(nx);
41 sub(Mxbj, x_next); // - x(j+1)
42 } else {
43 ctx.wait(std::move(arrival)); // x_next comes from next thread
44 auto x_next = x.batch(d1_next).bottom_rows(nx);
45 if (c_next > 0 || v == 1)
46 sub(Mxbj, x_next);
47 else
48 sub(Mxbj, x_next, with_rotate<1>);
49 }
50 }
51 // print_grad(ctx, b,"res_b");
52 // print_grad(ctx, x,"x");
53 // print_grad(ctx, Mxb,"Mxb");
54}

56template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
57void CyqloneSolver<VL, T, DefaultOrder, Ctx>::transposed_dynamics_constr(Context &ctx, view<> λ,
58 mut_view<> Mᵀλ,
59 bool accum) const {
60 // (Mᵀλ)(j) = [ B(j)ᵀ ] λ(j) - [ 0 ] λ(j-1)
61 // [ A(j)ᵀ ] [ I ]
62 auto arrival = ctx.arrive();
63 const index_t c = riccati_thread_assignment(ctx);
64 const index_t dn = c * n; // data batch index
65 const index_t c_prev = sub_wrap_p(c, 1);
66 const index_t dn_prev = c_prev * n;
67 for (index_t i = 0; i < n; ++i) {
68 const index_t j = stage_index(c, i, 0);
69 GUANAQO_TRACE("trans_dyn_constr", j);
70 index_t di = dn + i;
71 auto BAj = data_F.batch(di), Bj = BAj.left_cols(nu);
72 auto λj = λ.batch(di);
73 auto Mᵀλj = Mᵀλ.batch(di);
74 if (v == 1 && j == 0) {
75 accum ? gemv_add(Bj.transposed(), λj, Mᵀλj.top_rows(nu)) //
76 : gemv(Bj.transposed(), λj, Mᵀλj.top_rows(nu));
77 if (!accum)
78 Mᵀλj.bottom_rows(nx).set_constant(0);
79 } else {
80 accum ? gemv_add(BAj.transposed(), λj, Mᵀλj) //
81 : gemv(BAj.transposed(), λj, Mᵀλj);
82 }
83 if (i + 1 < n) {
84 index_t di_prev = di + 1; // j - 1
85 auto λ_prev = λ.batch(di_prev);
86 sub(Mᵀλj.bottom_rows(nx), λ_prev);
87 } else {
88 ctx.wait(std::move(arrival)); // λ_prev comes from previous thread
89 auto λ_prev = λ.batch(dn_prev);
90 if (c > 0 || v == 1)
91 sub(Mᵀλj.bottom_rows(nx), λ_prev);
92 else
93 sub(Mᵀλj.bottom_rows(nx), λ_prev, with_rotate<-1>);
94 }
95 }

132template <index_t VL, class T, StorageOrder DefaultOrder, class Ctx>
133void CyqloneSolver<VL, T, DefaultOrder, Ctx>::cost_gradient_regularized(Context &ctx, value_type γ,
134 view<> ux, view<> ux0,
135 view<> q,
136 mut_view<> grad_f) const {
137 simd inv_γ{1 / γ};
138 const auto reg_simd = [inv_γ](auto qji, auto xji, auto x0ji) {
139 return inv_γ * (xji - x0ji) + qji;
140 };
141 const auto mul_Hx = [&]([[maybe_unused]] auto j, auto, auto qj, auto Hj, auto uxj, auto ux0j,
142 auto grad_fj) {
143 GUANAQO_TRACE("cost_gradient_regularized", j);
144 linalg::transform_elementwise(reg_simd, grad_fj, qj, uxj, ux0j);
145 symv_add(tril(Hj), uxj, grad_fj);
146 };

151void CyqloneSolver<VL, T, DefaultOrder, Ctx>::cost_gradient_remove_regularization(
152 Context &ctx, value_type γ, view<> ux, view<> ux0, mut_view<> grad_f) const {
153 simd inv_γ{1 / γ};
154 const auto sub_reg_simd = [inv_γ](auto grad_fji, auto xji, auto x0ji) {
155 return grad_fji + inv_γ * (x0ji - xji);
